set -euo pipefail

# --- 0) Vars you can change ---
KIOSK_URL="${KIOSK_URL:-https://itank.io/}"
KIOSK_REFRESH_SECS="${KIOSK_REFRESH_SECS:-0}"   # 0 = no periodic refresh
NET_WAIT_SECS="${NET_WAIT_SECS:-45}"
GPU_SPLIT_MB="${GPU_SPLIT_MB:-64}"
SWAP_DISK_MB="${SWAP_DISK_MB:-256}"
USER_NAME="$(whoami)"
LOG_FILE="/var/log/kiosk-xinit.log"

echo "[i] Preparing system (console boot, GPU split, packages)…"

# --- 1) Boot to Console Autologin (required) ---
sudo raspi-config nonint do_boot_behaviour B2 || true   # Console Autologin

# --- 2) Disable blanking and set GPU split low (saves RAM) ---
sudo raspi-config nonint do_blanking 0 || true
sudo raspi-config nonint do_memory_split "${GPU_SPLIT_MB}" || true

# --- 3) Install minimal X + Chromium (either name) + helpers ---
sudo apt update
sudo apt -y install --no-install-recommends xserver-xorg xinit openbox unclutter xdotool wmctrl zram-tools dphys-swapfile || true
if ! command -v chromium-browser >/dev/null 2>&1; then
  sudo apt -y install chromium || true
fi
if ! command -v chromium-browser >/dev/null 2>&1 && ! command -v chromium >/dev/null 2>&1; then
  echo "[!] Chromium package not found. Try: sudo apt search chromium"
  exit 1
fi

# --- 4) zram + tiny swap (prevents OOM on 512MB) ---
if [ -f /etc/ztab ]; then sudo cp /etc/ztab /etc/ztab.bak.$(date +%s); fi
sudo tee /etc/ztab >/dev/null <<'ZTAB'
swap   swap  100  ram/2  0  lz4  1
tmp    zram  0    100M   0  lz4  1  /tmp  ext4  rw,noatime,mode=1777
ZTAB
sudo systemctl enable --now zram-config || sudo systemctl restart zram-config || true
sudo sed -i "s/^CONF_SWAPSIZE=.*/CONF_SWAPSIZE=${SWAP_DISK_MB}/" /etc/dphys-swapfile || true
sudo systemctl enable dphys-swapfile
sudo systemctl restart dphys-swapfile || true

# --- 5) Minimal Openbox config ---
mkdir -p "$HOME/.config/openbox"
cat > "$HOME/.config/openbox/autostart" <<'OB'
unclutter -idle 1 -root &
OB

# --- 6) Chromium auto-login extension directory (optional; safe even if unused) ---
mkdir -p "$HOME/kiosk-ext"

# --- 7) kiosk-xinit.sh (runs Chromium in lean mode) ---
sudo tee /usr/local/bin/kiosk-xinit.sh >/dev/null <<'BASH'
#!/usr/bin/env bash
set -euo pipefail
URL="${KIOSK_URL:-https://itank.io/}"
REFRESH_SECS="${KIOSK_REFRESH_SECS:-0}"
NET_WAIT_SECS="${NET_WAIT_SECS:-45}"

# Wait for network
for i in $(seq 1 "$NET_WAIT_SECS"); do
  if ping -c1 -W1 1.1.1.1 >/dev/null 2>&1 || ping -c1 -W1 google.com >/dev/null 2>&1; then break; fi
  sleep 1
done

xset s off; xset -dpms; xset s noblank
unclutter -idle 1 -root &
openbox-session &

if [[ "$REFRESH_SECS" != "0" ]]; then
  ( while true; do sleep "$REFRESH_SECS"; command -v xdotool >/dev/null 2>&1 && xdotool key Ctrl+R; done ) &
fi

CHROMIUM_BIN="$(command -v chromium-browser || true)"
[[ -z "$CHROMIUM_BIN" ]] && CHROMIUM_BIN="$(command -v chromium || true)"
[[ -z "$CHROMIUM_BIN" ]] && { echo "Chromium not found"; exit 1; }

exec "$CHROMIUM_BIN" \
  --kiosk \
  --start-fullscreen \
  --no-first-run \
  --no-default-browser-check \
  --noerrdialogs \
  --disable-session-crashed-bubble \
  --disable-infobars \
  --disable-translate \
  --disable-background-networking \
  --disable-backgrounding-occluded-windows \
  --disable-component-update \
  --disable-domain-reliability \
  --disable-features=RendererCodeIntegrity,AutofillServerCommunication \
  --process-per-site \
  --renderer-process-limit=2 \
  --password-store=basic \
  --aggressive-cache-discard \
  --disk-cache-size=15728640 \
  --media-cache-size=10485760 \
  --overscroll-history-navigation=0 \
  --autoplay-policy=no-user-gesture-required \
  --user-data-dir="$HOME/.kiosk-chrome" \
  --load-extension="$HOME/kiosk-ext" \
  "$URL"
BASH
sudo chmod +x /usr/local/bin/kiosk-xinit.sh

# --- 8) systemd unit: start X on tty1 and run kiosk-xinit.sh ---
sudo tee /etc/systemd/system/kiosk@.service >/dev/null <<'UNIT'
[Unit]
Description=Chromium Kiosk (X via startx) for user %i
After=systemd-user-sessions.service getty@tty1.service network-online.target
Wants=network-online.target

[Service]
User=%i
WorkingDirectory=/home/%i
TTYPath=/dev/tty1
Environment=KIOSK_URL=https://itank.io/
Environment=KIOSK_REFRESH_SECS=0
Environment=NET_WAIT_SECS=45
PAMName=login
Type=simple
StandardInput=tty
StandardOutput=journal
StandardError=journal
ExecStartPre=/bin/bash -lc 'pkill -u %i -f "startx" || true'
ExecStart=/bin/bash -lc 'startx /usr/local/bin/kiosk-xinit.sh -- :0 vt1'
Restart=on-failure
RestartSec=3

[Install]
WantedBy=multi-user.target
UNIT

# inject your custom env into the unit drop-in (so you can change URL without editing the unit)
sudo mkdir -p /etc/systemd/system/kiosk@${USER_NAME}.service.d
sudo tee /etc/systemd/system/kiosk@${USER_NAME}.service.d/override.conf >/dev/null <<EOF
[Service]
Environment=KIOSK_URL=${KIOSK_URL}
Environment=KIOSK_REFRESH_SECS=${KIOSK_REFRESH_SECS}
Environment=NET_WAIT_SECS=${NET_WAIT_SECS}
EOF

# --- 9) Enable + start the service ---
sudo systemctl daemon-reload
sudo systemctl enable kiosk@"${USER_NAME}".service
sudo systemctl stop lightdm 2>/dev/null || true
sudo systemctl restart kiosk@"${USER_NAME}".service

echo "[i] Kiosk service started. Check status with:"
echo "    systemctl status kiosk@${USER_NAME}.service --no-pager"
echo "    journalctl -u kiosk@${USER_NAME}.service -b --no-pager"
