cat > /tmp/remove-old-kiosk.sh <<'REMOVE_OLD_KIOSK'
#!/usr/bin/env bash
set -u -o pipefail

echo "=============================================="
echo "Removing old Raspberry Pi kiosk installation"
echo "=============================================="

if [ "$(id -u)" -ne 0 ]; then
  SUDO="sudo"
else
  SUDO=""
fi

run() {
  $SUDO "$@" 2>/dev/null || true
}

KIOSK_USER=""
KIOSK_HOME=""

if [ -f /etc/default/kiosk ]; then
  KIOSK_USER="$(grep '^KIOSK_USER=' /etc/default/kiosk | tail -n1 | cut -d= -f2- | sed "s/^'//;s/'$//;s/^\"//;s/\"$//")"
  KIOSK_HOME="$(grep '^KIOSK_HOME=' /etc/default/kiosk | tail -n1 | cut -d= -f2- | sed "s/^'//;s/'$//;s/^\"//;s/\"$//")"
fi

if [ -z "$KIOSK_USER" ]; then
  KIOSK_USER="${SUDO_USER:-pi}"
fi

if [ -z "$KIOSK_HOME" ]; then
  if getent passwd "$KIOSK_USER" >/dev/null 2>&1; then
    KIOSK_HOME="$(getent passwd "$KIOSK_USER" | cut -d: -f6)"
  else
    KIOSK_HOME="/home/$KIOSK_USER"
  fi
fi

BACKUP_DIR="/root/kiosk-remove-backup-$(date +%Y%m%d-%H%M%S)"
run mkdir -p "$BACKUP_DIR"

backup() {
  path="$1"
  if [ -e "$path" ] || [ -L "$path" ]; then
    run mkdir -p "$BACKUP_DIR$(dirname "$path")"
    run cp -a "$path" "$BACKUP_DIR$path"
  fi
}

remove_file() {
  backup "$1"
  run rm -f "$1"
}

remove_dir() {
  backup "$1"
  run rm -rf "$1"
}

echo
echo "[1/9] Stopping kiosk/browser/X processes..."

run pkill -TERM -u "$KIOSK_USER" -x chromium
run pkill -TERM -u "$KIOSK_USER" -x chromium-browser
run pkill -TERM -u "$KIOSK_USER" -x xinit
run pkill -TERM -u "$KIOSK_USER" -x Xorg
run pkill -TERM -u "$KIOSK_USER" -x openbox
run pkill -TERM -u "$KIOSK_USER" -x xbindkeys
run pkill -TERM -u "$KIOSK_USER" -x unclutter
sleep 2
run pkill -KILL -u "$KIOSK_USER" -x chromium
run pkill -KILL -u "$KIOSK_USER" -x chromium-browser

echo
echo "[2/9] Removing kiosk systemd services..."

run systemctl stop kiosk-guard.service
run systemctl disable kiosk-guard.service
run systemctl stop kiosk-auto-reboot.timer
run systemctl disable kiosk-auto-reboot.timer
run systemctl stop kiosk-auto-reboot.service
run systemctl disable kiosk-auto-reboot.service
run systemctl reset-failed kiosk-guard.service
run systemctl reset-failed kiosk-auto-reboot.timer
run systemctl reset-failed kiosk-auto-reboot.service

remove_file /etc/systemd/system/kiosk-guard.service
remove_file /etc/systemd/system/kiosk-auto-reboot.service
remove_file /etc/systemd/system/kiosk-auto-reboot.timer

run systemctl daemon-reload
run systemctl reset-failed

echo
echo "[3/9] Removing kiosk scripts..."

remove_file /usr/local/bin/kiosk-wifi.sh
remove_file /usr/local/bin/kiosk-xinit.sh
remove_file /usr/local/bin/kiosk-guard.sh
remove_file /etc/default/kiosk

echo
echo "[4/9] Removing Chrome kiosk profile and extension..."

remove_dir "$KIOSK_HOME/.kiosk-chrome"
remove_dir "$KIOSK_HOME/kiosk-ext"

echo
echo "[5/9] Removing Openbox and hotkey kiosk configs..."

remove_file "$KIOSK_HOME/.xbindkeysrc"

if [ -f "$KIOSK_HOME/.config/openbox/autostart" ]; then
  if grep -q "unclutter -idle 1 -root" "$KIOSK_HOME/.config/openbox/autostart"; then
    remove_file "$KIOSK_HOME/.config/openbox/autostart"
  fi
fi

echo
echo "[6/9] Removing kiosk autostart from .bash_profile..."

PROFILE="$KIOSK_HOME/.bash_profile"

if [ -f "$PROFILE" ]; then
  backup "$PROFILE"
  TMP_PROFILE="$(mktemp)"
  sed '/# BEGIN KIOSK AUTOSTART/,/# END KIOSK AUTOSTART/d' "$PROFILE" > "$TMP_PROFILE"

  if id "$KIOSK_USER" >/dev/null 2>&1; then
    KIOSK_GROUP="$(id -gn "$KIOSK_USER")"
    run install -o "$KIOSK_USER" -g "$KIOSK_GROUP" -m 0644 "$TMP_PROFILE" "$PROFILE"
  else
    run install -m 0644 "$TMP_PROFILE" "$PROFILE"
  fi

  rm -f "$TMP_PROFILE"
fi

echo
echo "[7/9] Removing watchdog configs..."

remove_file /etc/modules-load.d/kiosk-watchdog.conf
remove_file /etc/systemd/system.conf.d/10-kiosk-watchdog.conf
run modprobe -r bcm2835_wdt
run systemctl daemon-reexec

echo
echo "[8/9] Removing GPU/ZRAM kiosk configs..."

for CFG in /boot/config.txt /boot/firmware/config.txt; do
  if [ -f "$CFG" ]; then
    backup "$CFG"
    run sed -i '/^gpu_mem=/d' "$CFG"
  fi
done

remove_file /etc/systemd/zram-generator.conf

if ls /etc/ztab.bak.* >/dev/null 2>&1; then
  LATEST_ZTAB_BACKUP="$(ls -1t /etc/ztab.bak.* | head -n1)"
  if [ -n "$LATEST_ZTAB_BACKUP" ]; then
    backup /etc/ztab
    run cp "$LATEST_ZTAB_BACKUP" /etc/ztab
  fi
else
  if [ -f /etc/ztab ]; then
    if grep -q "ram/3" /etc/ztab && grep -q "lz4" /etc/ztab; then
      remove_file /etc/ztab
    fi
  fi
fi

run systemctl daemon-reload

echo
echo "[9/9] Removing kiosk-related packages except NetworkManager..."

run apt-get purge -y \
  xserver-xorg \
  xinit \
  openbox \
  unclutter \
  xdotool \
  wmctrl \
  xbindkeys \
  xterm \
  dbus-x11 \
  x11-xserver-utils \
  xserver-xorg-legacy \
  chromium \
  chromium-browser \
  systemd-zram-generator \
  zram-tools

run apt-get autoremove -y

echo
echo "Restoring normal boot setting if raspi-config exists..."

if command -v raspi-config >/dev/null 2>&1; then
  run raspi-config nonint do_boot_behaviour B1
  run raspi-config nonint do_blanking 1
fi

remove_file /etc/Xwrapper.config

run systemctl daemon-reload
run systemctl reset-failed

echo
echo "=============================================="
echo "Old kiosk removed successfully."
echo "Backup saved at: $BACKUP_DIR"
echo "Now reboot the Pi:"
echo "sudo reboot"
echo "=============================================="
REMOVE_OLD_KIOSK

sed -i 's/\r$//' /tmp/remove-old-kiosk.sh
chmod +x /tmp/remove-old-kiosk.sh
sudo bash /tmp/remove-old-kiosk.sh
