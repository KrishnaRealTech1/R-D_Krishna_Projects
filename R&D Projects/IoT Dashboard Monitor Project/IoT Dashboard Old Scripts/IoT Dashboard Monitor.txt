Steps:

1) Flash the OS (on your PC/Mac)

*Use Raspberry Pi Imager → choose Raspberry Pi OS (32-bit) with Desktop.

*Click the gear icon:

    Set username/password

    Set Wi-Fi SSID/Password + Country

    Set Timezone

*(Optional) Enable SSH

*Flash → insert the card → boot the Pi.


DESKTOP AUTO-LOGIN IS ENABLED:
Menu → Preferences → Raspberry Pi Configuration → System → Boot: To Desktop, Login: Auto login.


2) On the Pi, run the one-time setup script

*Press Ctrl + Alt + T (opens Terminal instantly).

FROM THE MENU
*Click the Raspberry (Start) icon → Accessories → Terminal (sometimes labeled LXTerminal).

RIGHT-CLICK
Right-click on the desktop → Open in Terminal (if available).



*Open Terminal on the Pi and paste this whole block:

#!/usr/bin/env bash
set -euo pipefail
USER_HOME="$HOME"
USER_NAME="$(whoami)"

echo "== Updating system and installing packages =="
sudo apt update
sudo apt -y full-upgrade

  "name": "Kiosk Auto Login (itank.io)",
  "version": "1.0.0",

  "permissions": ["storage"],
  "host_permissions": ["https://itank.io/*"],

cat > "$USER_HOME/kiosk-ext/content.js" << 'JS'
// Kiosk Auto Login for itank.io
(function () {
  const STORAGE_KEY = "itankLastCreds";
  const AUTOSUBMIT_DELAY_MS = 1200;
  const SEARCH_TIMEOUT_MS = 8000;
  const RESCAN_INTERVAL_MS = 1500;
  let cancelAuto = false, submitTimer = null, attached = false;

  function setInputValue(el, val) {
    if (!el) return;
    const d = Object.getOwnPropertyDescriptor(Object.getPrototypeOf(el), 'value');
    if (d && d.set) d.set.call(el, val); else el.value = val;
    el.dispatchEvent(new Event('input', { bubbles: true }));
    el.dispatchEvent(new Event('change', { bubbles: true }));
  }
  function looksLikeUsername(el) {
    const name = (el.name||"").toLowerCase();
    const id = (el.id||"").toLowerCase();
    const ph = (el.placeholder||"").toLowerCase();
    const type = (el.type||"").toLowerCase();
    return (type==="text"||type==="email"||name.includes("user")||name.includes("email")||name.includes("login")||id.includes("user")||id.includes("email")||id.includes("login")||ph.includes("user")||ph.includes("email")||ph.includes("login"));
  }
  function findLoginForm() {
    const pass = document.querySelector('input[type="password"]');
    if (!pass) return null;
    let form = pass.closest("form") || pass.closest('[role="form"]') || pass.closest("div");
    const candidates = (form||document).querySelectorAll('input[type="text"], input[type="email"], input[name], input[id], input[placeholder]');
    let user = null;
    for (const el of candidates) { if (el!==pass && looksLikeUsername(el)) { user = el; break; } }
    if (!user) {
      const arr = Array.from((form||document).querySelectorAll('input'));
      const idx = arr.indexOf(pass);
      if (idx>0) { const prev = arr[idx-1]; if (prev && prev.type!=="password") user = prev; }
    }
    let submitBtn = (form && form.querySelector('button[type="submit"], input[type="submit"]')) || (form && form.querySelector('button, input[type="button"]')) || document.querySelector('button[type="submit"], input[type="submit"]');
    return { form, user, pass, submitBtn };
  }
  function banner() {
    const b = document.createElement("div");
    b.textContent = "Auto-login in 1.2s — press Esc to cancel";
    Object.assign(b.style, { position:"fixed", right:"8px", bottom:"8px", background:"rgba(0,0,0,0.75)", color:"#fff", padding:"6px 10px", borderRadius:"6px", font:"12px/1.2 sans-serif", zIndex:2147483647, pointerEvents:"none" });
    document.body.appendChild(b);
    const onKey = (e)=>{ if(e.key==="Escape"){ cancelAuto=true; if(submitTimer) clearTimeout(submitTimer); b.textContent="Auto-login canceled"; setTimeout(()=>b.remove(),1200); window.removeEventListener("keydown", onKey, true);} };
    window.addEventListener("keydown", onKey, true);
    setTimeout(()=>b.remove(), AUTOSUBMIT_DELAY_MS+1500);
  }
  function loadCreds(){ return new Promise(r=>chrome.storage.local.get(STORAGE_KEY, d=>r(d[STORAGE_KEY]||null))); }
  function saveCreds(u,p){ return new Promise(r=>{ const x={}; x[STORAGE_KEY]={username:u,password:p,ts:Date.now()}; chrome.storage.local.set(x,()=>r()); }); }
  function submit(nodes){
    const {form, submitBtn, pass} = nodes;
    if (form && typeof form.requestSubmit==="function") form.requestSubmit();
    else if (submitBtn && typeof submitBtn.click==="function") submitBtn.click();
    else if (form && typeof form.submit==="function") form.submit();
    else { pass.dispatchEvent(new KeyboardEvent("keydown",{key:"Enter",bubbles:true})); pass.dispatchEvent(new KeyboardEvent("keyup",{key:"Enter",bubbles:true})); }
  }
  function wireCapture(nodes){
    if (attached) return;
    const {form, user, pass} = nodes;
    const h = ()=>{ const u=user?user.value:"", p=pass?pass.value:""; if(u && p) saveCreds(u,p); };
    if (form) form.addEventListener("submit", h, {capture:true});
    else { document.addEventListener("click", h, {capture:true}); pass&&pass.addEventListener("keydown", e=>{if(e.key==="Enter") h();}, {capture:true}); }
    attached = true;
  }
  async function tryOnce(){
    const t0 = Date.now();
    while (Date.now()-t0 < SEARCH_TIMEOUT_MS) {
      const nodes = findLoginForm();
      if (!nodes || !nodes.pass) return;
      const {user, pass} = nodes;
      const creds = await loadCreds();
      if (user && creds && creds.username && creds.password) {
        setInputValue(user, creds.username);
        setInputValue(pass, creds.password);
        wireCapture(nodes);
        banner();
        submitTimer = setTimeout(()=>{ if(!cancelAuto) submit(nodes); }, AUTOSUBMIT_DELAY_MS);
        return;
      }
      wireCapture(nodes);
      return;
    }
  }
  tryOnce();
  const iv = setInterval(()=>{ tryOnce().catch(()=>{}); }, RESCAN_INTERVAL_MS);
  setTimeout(()=>clearInterval(iv), 60000);
})();
JS

cat > "$USER_HOME/bin/start-kiosk.sh" << 'BASH'
#!/usr/bin/env bash
# -------- Kiosk launcher (Desktop autostart) --------
set -euo pipefail

URL="${KIOSK_URL:-https://itank.io/}"
REFRESH_SECS="${KIOSK_REFRESH_SECS:-0}"
NET_WAIT_SECS="${NET_WAIT_SECS:-60}"

# Wait for network (default route or DNS working)
for i in $(seq 1 "$NET_WAIT_SECS"); do
  if ping -c1 -W1 1.1.1.1 >/dev/null 2>&1 || ping -c1 -W1 google.com >/dev/null 2>&1; then
    break
  fi
  sleep 1
done

# Stop display sleep/blanking and hide cursor
xset s off
xset -dpms
xset s noblank
unclutter -idle 1 -root &

# Optional periodic reload (Ctrl+R) if REFRESH_SECS > 0
if [[ "$REFRESH_SECS" != "0" ]]; then
  (
    while true; do
      sleep "$REFRESH_SECS"
      if command -v wmctrl >/dev/null 2>&1 && command -v xdotool >/dev/null 2>&1; then
        wmctrl -x -a "chromium" && xdotool key Ctrl+R
      fi
    done
  ) &
fi

# === Chromium kiosk; relaunch if it exits ===
while true; do
  chromium-browser \
    --kiosk \
    --start-fullscreen \
    --noerrdialogs \
    --disable-session-crashed-bubble \
    --disable-infobars \
    --disable-translate \
    --check-for-update-interval=31536000 \
    --overscroll-history-navigation=0 \
    --password-store=basic \
    --autoplay-policy=no-user-gesture-required \
    --user-data-dir="$HOME/.kiosk-chrome" \
    --load-extension="$HOME/kiosk-ext" \
    "$URL"
  sleep 1
done
BASH

chmod +x "$USER_HOME/bin/start-kiosk.sh"
chown -R "$USER_NAME:$USER_NAME" "$USER_HOME/kiosk-ext" "$USER_HOME/.config" "$USER_HOME/bin"
echo "== Done. Rebooting in 3s… Press Ctrl+C to cancel =="
sleep 3
sudo reboot
EOF
