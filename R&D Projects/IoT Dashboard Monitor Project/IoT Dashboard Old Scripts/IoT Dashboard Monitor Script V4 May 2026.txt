set -euo pipefail
# ============================================================================
# Universal Raspberry Pi Console Kiosk (Future-proof + Watchdog)
# - Console autologin -> X -> Chromium kiosk (itank.io)
# - Chromium opens immediately, even if network/internet is not connected
# - Auto-login with last-used credentials (+ Reset link)
# - Ctrl+Alt+W -> Wi-Fi (nmtui) in fullscreen xterm; kiosk restarts after exit
# - NetworkManager preferred; dhcpcd disabled if conflicting
# - ZRAM via systemd-zram-generator if present, else zram-tools (/etc/ztab)
# - Safe gpu_mem config for /boot/config.txt and /boot/firmware/config.txt

# - Optional systemd hardware watchdog

KIOSK_URL="${KIOSK_URL:-https://web.itank.io/login/}"
KIOSK_REFRESH_SECS="${KIOSK_REFRESH_SECS:-300}"    # 0 = off; 300s = 5 minutes

run_root() {
  if [[ ${#SUDO[@]} -gt 0 ]]; then
    "${SUDO[@]}" "$@"

    "$@"
  fi

write_root_file() {
  local path="$1"
  local mode="${2:-0644}"
  local tmp
  tmp="$(mktemp)"
  cat >"$tmp"
  run_root install -D -m "$mode" "$tmp" "$path"
  rm -f "$tmp"
}

(function () {
  const KEY = "itankLastCreds";
  const DELAY = 1200;
  const RESCAN = 1500;
  let cancel = false;
  let attachedForm = null;
  let timer = null;
  let autoAttempted = false;
  let bannerVisible = false;

  const load = () => new Promise((resolve) => chrome.storage.local.get(KEY, (data) => resolve(data[KEY] || null)));
  const save = (username, password) => new Promise((resolve) => {
    const data = {};
    data[KEY] = { username, password, ts: Date.now() };
    chrome.storage.local.set(data, () => resolve());
  });
  const clear = () => new Promise((resolve) => chrome.storage.local.remove(KEY, () => resolve()));

  function setVal(el, value) {
    if (!el) return;
    const descriptor = Object.getOwnPropertyDescriptor(Object.getPrototypeOf(el), "value");
    if (descriptor && descriptor.set) descriptor.set.call(el, value);
    else el.value = value;
    el.dispatchEvent(new Event("input", { bubbles: true }));
    el.dispatchEvent(new Event("change", { bubbles: true }));
  }

  function looksUser(el) {
    const name = (el.name || "").toLowerCase();
    const id = (el.id || "").toLowerCase();
    const placeholder = (el.placeholder || "").toLowerCase();
    const type = (el.type || "").toLowerCase();
    return type === "text" || type === "email" ||
      name.includes("user") || name.includes("email") || name.includes("login") ||
      id.includes("user") || id.includes("email") || id.includes("login") ||
      placeholder.includes("user") || placeholder.includes("email") || placeholder.includes("login");
  }

  function findLoginFields() {
    const pass = document.querySelector('input[type="password"]');
    if (!pass) return null;
    const form = pass.closest("form") || pass.closest('[role="form"]') || pass.closest("div");
    const inputs = Array.from((form || document).querySelectorAll("input"));
    const user = inputs.find((el) => el !== pass && looksUser(el)) || null;
    const submit = (form && form.querySelector('button[type="submit"],input[type="submit"]')) ||
      document.querySelector('button[type="submit"],input[type="submit"]');
    return { form, user, pass, submit };
  }

  function showBanner(hasCreds) {
    if (bannerVisible || !document.body) return;
    bannerVisible = true;
    const b = document.createElement("div");
    b.innerHTML = hasCreds
      ? 'Auto-login in 1.2s — press <b>Esc</b> to cancel | <b id="resetCreds">Reset</b>'
      : 'No saved credentials';
    Object.assign(b.style, {
      position: "fixed",
      right: "8px",
      bottom: "8px",
      background: "rgba(0,0,0,0.75)",
      color: "#fff",
      padding: "6px 10px",
      borderRadius: "6px",
      font: "12px/1.2 sans-serif",
      zIndex: 2147483647
    });
    document.body.appendChild(b);

    const reset = b.querySelector("#resetCreds");
    if (reset) {
      reset.style.cursor = "pointer";
      reset.onclick = () => {
        clear().then(() => {
          cancel = true;
          clearTimeout(timer);
          b.textContent = "Credentials cleared";
          setTimeout(() => b.remove(), 1200);
        });
      };
    }

    window.addEventListener("keydown", (event) => {
      if (event.key === "Escape") {
        cancel = true;
        clearTimeout(timer);
        b.textContent = "Auto-login canceled";
        setTimeout(() => b.remove(), 1200);
      }
    }, { once: true });

    setTimeout(() => {
      b.remove();
      bannerVisible = false;
    }, DELAY + 2000);
  }

  function doSubmit(nodes) {
    const { form, submit, pass } = nodes;
    if (form && form.requestSubmit) form.requestSubmit();
    else if (submit && submit.click) submit.click();
    else if (form && form.submit) form.submit();
    else {
      pass.dispatchEvent(new KeyboardEvent("keydown", { key: "Enter", bubbles: true }));
      pass.dispatchEvent(new KeyboardEvent("keyup", { key: "Enter", bubbles: true }));
    }
  }

  function wire(nodes) {
    const { form, user, pass, submit } = nodes;
    if (attachedForm === form) return;

    const persist = () => {
      const username = user ? user.value.trim() : "";
      const password = pass ? pass.value : "";
      if (username && password) save(username, password).catch(() => {});
    };

    if (form) form.addEventListener("submit", persist, { capture: true });
    if (submit) submit.addEventListener("click", persist, { capture: true });
    if (pass) pass.addEventListener("change", persist, { capture: true });
    attachedForm = form;
  }

  async function once() {
    const nodes = findLoginFields();
    if (!nodes || !nodes.pass) return;
    const { user, pass } = nodes;
    wire(nodes);

    if (autoAttempted) return;
    const creds = await load();
    if (creds && creds.username && creds.password) {
      autoAttempted = true;
      setVal(user, creds.username);
      setVal(pass, creds.password);
      showBanner(true);
      timer = setTimeout(() => {
        if (!cancel) doSubmit(nodes);
      }, DELAY);
      return;
    }
    showBanner(false);
  }

  once().catch(() => {});
  const iv = setInterval(() => once().catch(() => {}), RESCAN);
  setTimeout(() => clearInterval(iv), 60000);
})();
JS

set -euo pipefail
if [[ -f /etc/default/kiosk ]]; then

[[ -z "$CHROMIUM_BIN" ]] && { echo "Chromium not found"; exit 1; }

  --noerrdialogs --disable-session-crashed-bubble --disable-infobars \
  --disable-translate --disable-background-networking \
  --disable-backgrounding-occluded-windows --disable-component-update \
  --disable-domain-reliability \
  --disable-features=AutofillServerCommunication \
  --process-per-site --renderer-process-limit=2 \
  --password-store=basic --aggressive-cache-discard \
  --disk-cache-size=15728640 --media-cache-size=10485760 \
  --overscroll-history-navigation=0 --autoplay-policy=no-user-gesture-required \
  --user-data-dir="$USER_HOME/.kiosk-chrome" --load-extension="$USER_HOME/kiosk-ext" \
  "$URL"

# ---------- 9) Keepalive + software watchdog service ----------
write_root_file /usr/local/bin/kiosk-guard.sh 0755 <<'BASH'
#!/usr/bin/env bash
set -u -o pipefail
if [[ -f /etc/default/kiosk ]]; then
  # shellcheck disable=SC1091
  . /etc/default/kiosk
fi

KIOSK_USER="${KIOSK_USER:-pi}"
ENABLE_KEEPALIVE="${ENABLE_KEEPALIVE:-1}"
KEEPALIVE_HOST="${KEEPALIVE_HOST:-google.com}"
KEEPALIVE_FALLBACK_HOST="${KEEPALIVE_FALLBACK_HOST:-1.1.1.1}"
KEEPALIVE_INTERVAL_SECS="${KEEPALIVE_INTERVAL_SECS:-15}"
KEEPALIVE_FAIL_THRESHOLD="${KEEPALIVE_FAIL_THRESHOLD:-8}"
KEEPALIVE_ACTION_COOLDOWN_SECS="${KEEPALIVE_ACTION_COOLDOWN_SECS:-300}"
KEEPALIVE_RESTART_KIOSK_ON_FAILURE="${KEEPALIVE_RESTART_KIOSK_ON_FAILURE:-0}"

log() {
  echo "[$(date -Is)] $*"
}

notify_systemd() {
  if command -v systemd-notify >/dev/null 2>&1; then
    systemd-notify "$@" || true
  fi
}

restart_kiosk_session() {
  [[ "$KEEPALIVE_RESTART_KIOSK_ON_FAILURE" == "1" ]] || return 0
  log "Restarting kiosk browser session"
  pkill -TERM -u "$KIOSK_USER" -x chromium || true
  pkill -TERM -u "$KIOSK_USER" -x chromium-browser || true
  sleep 5
  pkill -KILL -u "$KIOSK_USER" -x chromium || true
  pkill -KILL -u "$KIOSK_USER" -x chromium-browser || true
}

recover_network() {
  log "Recovering network via NetworkManager"
  systemctl restart NetworkManager.service || true
  nmcli networking off || true
  sleep 2
  nmcli networking on || true
}

ping_ok() {
  ping -c1 -W2 "$KEEPALIVE_HOST" >/dev/null 2>&1 || ping -c1 -W2 "$KEEPALIVE_FALLBACK_HOST" >/dev/null 2>&1
}

notify_systemd --ready --status="kiosk guard running"
log "kiosk guard started: keepalive=${ENABLE_KEEPALIVE}, host=${KEEPALIVE_HOST}, interval=${KEEPALIVE_INTERVAL_SECS}s, restart_browser=${KEEPALIVE_RESTART_KIOSK_ON_FAILURE}"

fail_count=0
last_action=0

while true; do
  now="$(date +%s)"

  if [[ "$ENABLE_KEEPALIVE" != "1" ]]; then
    notify_systemd WATCHDOG=1 STATUS="keepalive disabled; watchdog alive"
    sleep "$KEEPALIVE_INTERVAL_SECS"
    continue
  fi

  if ping_ok; then
    if [[ "$fail_count" -gt 0 ]]; then
      log "keepalive restored after ${fail_count} failed check(s)"
    fi
    fail_count=0
    notify_systemd WATCHDOG=1 STATUS="keepalive ok: ${KEEPALIVE_HOST}"
  else
    fail_count=$((fail_count + 1))
    log "keepalive failed ${fail_count}/${KEEPALIVE_FAIL_THRESHOLD}: ${KEEPALIVE_HOST}"
    notify_systemd WATCHDOG=1 STATUS="keepalive failed ${fail_count}/${KEEPALIVE_FAIL_THRESHOLD}"

    if [[ "$fail_count" -ge "$KEEPALIVE_FAIL_THRESHOLD" ]]; then
      if [[ $((now - last_action)) -ge "$KEEPALIVE_ACTION_COOLDOWN_SECS" ]]; then
        last_action="$now"
        recover_network
        restart_kiosk_session
      else
        log "Recovery skipped due to cooldown"
      fi
      fail_count=0
    fi
  fi

  sleep "$KEEPALIVE_INTERVAL_SECS"
done
BASH

if [[ "$ENABLE_SOFTWARE_WATCHDOG" == "1" ]]; then
  WATCHDOG_LINE="WatchdogSec=${SOFTWARE_WATCHDOG_DURATION}"
else
  WATCHDOG_LINE=""
fi

After=NetworkManager.service multi-user.target
Wants=NetworkManager.service
StartLimitIntervalSec=300
StartLimitBurst=20

[Service]
Type=notify
NotifyAccess=all
ExecStart=/usr/local/bin/kiosk-guard.sh
Restart=always

${WATCHDOG_LINE}

# ---------- 10) Hardware watchdog option ----------
if [[ "$ENABLE_HW_WATCHDOG" == "1" ]]; then
  write_root_file /etc/modules-load.d/kiosk-watchdog.conf 0644 <<'EOF'
bcm2835_wdt
EOF
  run_root modprobe bcm2835_wdt || true
  write_root_file /etc/systemd/system.conf.d/10-kiosk-watchdog.conf 0644 <<EOF
[Manager]
RuntimeWatchdogSec=${HW_WATCHDOG_RUNTIME}
RebootWatchdogSec=${HW_WATCHDOG_REBOOT}
EOF
  echo "[i] Hardware watchdog config enabled. Takes full effect after reboot."
else
  run_root rm -f /etc/modules-load.d/kiosk-watchdog.conf /etc/systemd/system.conf.d/10-kiosk-watchdog.conf || true
  echo "[i] Hardware watchdog config disabled."
fi

# ---------- 11) Scheduled auto reboot every N hours ----------
SHUTDOWN_BIN="$(command -v shutdown || true)"
if [[ -z "$SHUTDOWN_BIN" ]]; then
  SHUTDOWN_BIN="/usr/sbin/shutdown"
fi

write_root_file /etc/systemd/system/kiosk-auto-reboot.service 0644 <<EOF
[Unit]
Description=Scheduled kiosk reboot

[Service]
Type=oneshot
ExecStart=${SHUTDOWN_BIN} -r now "Scheduled kiosk reboot every ${AUTO_REBOOT_HOURS} hour(s)"
EOF

write_root_file /etc/systemd/system/kiosk-auto-reboot.timer 0644 <<EOF
[Unit]
Description=Run scheduled kiosk reboot every ${AUTO_REBOOT_HOURS} hour(s)

[Timer]
OnActiveSec=${AUTO_REBOOT_HOURS}h
OnUnitActiveSec=${AUTO_REBOOT_HOURS}h
AccuracySec=1min
Persistent=false
Unit=kiosk-auto-reboot.service

[Install]
WantedBy=timers.target
EOF

# ---------- 12) Autostart from tty1 login shell ----------
PROFILE="$TARGET_HOME/.bash_profile"
if [[ -f "$PROFILE" ]]; then
  run_root cp "$PROFILE" "$PROFILE.bak.$(date +%s)"
else
  write_user_file "$PROFILE" 0644 <<'EOF'
EOF
fi

TMP_PROFILE="$(mktemp)"
run_root sed '/# BEGIN KIOSK AUTOSTART/,/# END KIOSK AUTOSTART/d' "$PROFILE" >"$TMP_PROFILE"
cat >>"$TMP_PROFILE" <<'EOF'

# BEGIN KIOSK AUTOSTART
# Auto-start kiosk when autologin lands on tty1 and X is not running.
if [ -z "${DISPLAY:-}" ] && [ "$(tty)" = "/dev/tty1" ]; then
  sleep 2
  exec startx /usr/local/bin/kiosk-xinit.sh -- :0 vt1
fi
# END KIOSK AUTOSTART
EOF
run_root install -o "$TARGET_USER" -g "$TARGET_GROUP" -m 0644 "$TMP_PROFILE" "$PROFILE"
rm -f "$TMP_PROFILE"

# ---------- 13) Enable services/timers ----------
run_root systemctl daemon-reload

if [[ "$ENABLE_KEEPALIVE" == "1" || "$ENABLE_SOFTWARE_WATCHDOG" == "1" ]]; then
  run_root systemctl enable --now kiosk-guard.service
else
  run_root systemctl disable --now kiosk-guard.service || true
fi

if [[ "$ENABLE_AUTO_REBOOT" == "1" ]]; then
  run_root systemctl enable --now kiosk-auto-reboot.timer
else
  run_root systemctl disable --now kiosk-auto-reboot.timer || true
fi

run_root systemctl daemon-reexec || true
