    def read_waste_types(self, filepath, encoding='utf-8'):
        try:
            with open(filepath, 'r', encoding=encoding) as file:
                return [line.strip() for line in file.readlines()]
        except FileNotFoundError:
            print(f"ENTER SOME WASTE TYPES IN WASTE_TYPES FILE")
            return []

    def read_credentials(self):
        credentials_path = os.path.join(os.path.expanduser("~"), "Desktop", "AWS_MICRO", "Credentials.txt")
        credentials = {}

        try:
            with open(credentials_path, 'r') as file:
                for line in file:
                    if '-' in line:
                        key, value = line.strip().split('-', 1)
                        credentials[key.strip()] = value.strip()
        except FileNotFoundError:
            print(f"Credentials file not found: {credentials_path}")

    def read_com_ports(self):
        com_ports = {"WASTE": None, "WEIGHT": None, "RFID": None}
        com_ports_file_path = os.path.join(os.path.expanduser("~"), "Desktop", "AWS_MICRO", "COM_PORT.txt")
        try:

        return com_ports

    # RFID Methods
    def run_rfid_reader(self, device_path, baud_rate=9600):
        try:
            ser = serial.Serial(device_path, baud_rate, timeout=1)
            while True:
                raw_data = ser.read(64)  # Read more bytes to handle fragmentation
                self.buffer += raw_data  # Append new data to buffer

                while len(self.buffer) >= 28:  # Process complete packets
                    packet = self.buffer[:28]  # Extract one full packet
                    self.buffer = self.buffer[28:]  # Remove processed data

                    tag_name_data = packet[10:22]  # Extract RFID tag
                    current_tag_name = self.parse_rfid_data(tag_name_data)

                    if (current_tag_name.startswith("E280") or current_tag_name.startswith(
                            "E200")) and current_tag_name != self.last_seen_tag:
                        print("New RFID tag detected:", current_tag_name)
                        self.master.after(0, lambda: self.rfid_value.set(current_tag_name))
                        self.master.after(0, lambda: self.update_vehicle_number_from_rfid(current_tag_name))
                        self.last_seen_tag = current_tag_name

        except serial.SerialException as e:
            print(f"Error opening or communicating with serial port: {e}")
        except Exception as e:
            print(f"Unexpected error in RFID reading or processing: {e}")

    def parse_rfid_data(self, raw_data):
        hex_str = ''.join(f"{byte:02X}" for byte in raw_data)
        return hex_str

    def update_vehicle_number_from_rfid(self, rfid_tag):
        url = "http://dev.igps.io/aws_madurai/api/rf_id_check.php"
        data = {"action": "get_vehicle", "rf_id": rfid_tag}
        try:
            response = requests.post(url, json=data)
            if response.ok:
                response_data = response.json()
                if response_data and 'vehicle_no' in response_data[0]:
                    vehicle_number = response_data[0].get('vehicle_no')
                else:
                    vehicle_number = "not.reg"

                self.vehicle_no_entry.delete(0, tk.END)
                self.vehicle_no_entry.insert(0, vehicle_number)
                print(vehicle_number)
            else:
                print(f"Server returned error code {response.status_code} for RFID {rfid_tag}")
                self.vehicle_no_entry.delete(0, tk.END)
                self.vehicle_no_entry.insert(0, "not.reg")
        except requests.RequestException as e:
            print(f"Network error when getting vehicle number for RFID {rfid_tag}: {e}")
            self.vehicle_no_entry.delete(0, tk.END)
            self.vehicle_no_entry.insert(0, "not.reg")

    # COM Port Methods
    def listen_to_com_port(self, port, baud_rate=9600, data_type='waste'):
        def handle_waste_data(data):
            if data in ["21", "24"]:
                self.master.after(0, self.handle_special_command, data)
            elif data == "22":
                self.master.after(0, self.clear_first_row)
            elif data == "23":
                self.master.after(0, self.clear_entire_table)
            else:
                try:
                    value = int(data)
                    if 1 <= value <= len(self.waste_types):
                        waste_type = self.waste_types[value - 1]
                        self.snapshot_filenames[waste_type] = []
                        self.master.after(0, lambda: self.update_table(sno=value, waste_type=waste_type))
                        camera_output_folder = os.path.join(os.path.expanduser("~"), "Desktop", "AWS_MICRO", "CAMERA_OUTPUT")
                        for i, rtsp_url in enumerate(self.rtsp_urls):
                            self.capture_and_save_snapshot(rtsp_url, i + 1, camera_output_folder,
                                                           lambda cam_id, filename: self.snapshot_filenames
                                                               [waste_type].append(filename) if filename else None)
                except ValueError:
                    print(f"Received non-integer value: {data}")

        def handle_weight_data(data):
            weight = self.extract_weight(data)
            self.master.after(0, lambda: self.update_table(weight=weight))
            self.master.after(0, lambda: self.weight_value.set(weight))

        try:
            with serial.Serial(port, baud_rate, timeout=2) as ser:
                while self.running:
                    if ser.in_waiting > 0:
                        data = ser.readline().decode(errors='ignore').strip()
                        if self.rfid_value.get():
                            if data_type == 'waste':
                                handle_waste_data(data)
                            elif data_type == 'weight':
                                handle_weight_data(data)
        except serial.SerialException as e:
            print(f"Serial port error: {e}")

    def start_com_threads(self):
        try:
            com_ports = self.read_com_ports()
            port_threads = {
                "RFID": (self.run_rfid_reader, 9600),
                "WASTE": (self.listen_to_com_port, 9600),
                "WEIGHT": (self.listen_to_com_port, 2400),
                #"WEIGHT": (self.listen_to_com_port, 9600)
            }

            for port_type, port_config in com_ports.items():
                if port_config and port_type in port_threads:
                    target, baud_rate = port_threads[port_type]
                    try:
                        if port_type == "RFID":
                            thread = threading.Thread(target=target, args=(port_config, baud_rate), daemon=True)
                        else:
                            thread = threading.Thread(target=target, args=(port_config, baud_rate, port_type.lower()), daemon=True)
                        thread.start()
                    except Exception as e:
                        print(f"Error starting thread for {port_type} on port {port_config}: {e}")
        except Exception as e:
            print(f"Error reading COM ports: {e}")

# Weight Data Methods
        # Weight Data Methods
        def extract_weight(self, weight_str):
            try:
                weight_str = weight_str.strip()

                # Check for format "wn" or "ww" prefix and "kg" suffix
                if (weight_str.startswith("wn") or weight_str.startswith("ww")) and weight_str.endswith("kg"):
                    weight_with_prefix = weight_str[2:-2]  # Remove "wn" or "ww" and "kg"
                    weight = float(weight_with_prefix)
                    return str(weight)

                # Check for a generic weight format (just a number)
                elif weight_str:  # Ensure it's not empty
                    weight = float(weight_str)
                    return str(weight)
                else:
                    return "0"  # Return "0" if empty string

            except ValueError as e:
                print(f"Error parsing weight from '{weight_str}': {e}")
                return "0"  # Return "0" if there's an error

    # Camera Methods
    def update_cameras(self):
        for i, rtsp_url in enumerate(self.rtsp_urls):
            threading.Thread(target=self.update_camera, args=(i, rtsp_url), daemon=True).start()

    def update_camera(self, index, rtsp_url):
        # Use FFmpeg as the backend for better performance
        cap = cv2.VideoCapture(rtsp_url, cv2.CAP_FFMPEG)

        # Reduce latency and buffer size
        cap.set(cv2.CAP_PROP_BUFFERSIZE, 1)
        cap.set(cv2.CAP_PROP_FPS, 15)  # Adjust the FPS as needed

        if not cap.isOpened():
            print(f"Error: Unable to open camera stream at {rtsp_url}")
            return

        while self.running:
            ret, frame = cap.read()

            if not ret or frame is None:
                print(f"Warning: Failed to read frame from {rtsp_url}")
                time.sleep(0.1)  # Small delay to avoid CPU overload
                continue

            try:
                # Convert color and resize frame
                frame = cv2.cvtColor(frame, cv2.COLOR_BGR2RGB)
                frame = cv2.resize(frame, (175, 175))

                # Convert to Tkinter-compatible image
                img = Image.fromarray(frame)
                img = ImageTk.PhotoImage(image=img)

                # Update camera feed in the UI
                self.cameras[index].configure(image=img)
                self.cameras[index].image = img

            except Exception as e:
                print(f"Error processing frame: {e}")
                break

        cap.release()
        print(f"Camera {index} at {rtsp_url} released.")

    def capture_and_save_snapshot(self, rtsp_url, camera_id, output_folder, callback):
        def capture():
            cap = cv2.VideoCapture(rtsp_url, cv2.CAP_FFMPEG)
            if not cap.isOpened():
                print(f"Error: Unable to open camera for snapshot at {rtsp_url}")
                callback(camera_id, None)
                return

            success, frame = cap.read()
            cap.release()

            if success and frame is not None:
                timestamp = time.strftime("%H%M%S_%d%m%y")
                filename = f"{self.credentials.get('IMG_NAME')}_{timestamp}_camera{camera_id}.jpg"
                filepath = os.path.join(output_folder, filename)

                try:
                    with concurrent.futures.ThreadPoolExecutor(max_workers=5) as executor:
                        future = executor.submit(cv2.imwrite, filepath, frame)
                        if future.result():
                            print(f"Snapshot saved: {filepath}")
                            callback(camera_id, filename)
                        else:
                            print(f"Failed to save snapshot: {filepath}")
                            callback(camera_id, None)
                except Exception as e:
                    print(f"Snapshot saving error: {e}")
                    callback(camera_id, None)
            else:
                print(f"Failed to capture snapshot from {rtsp_url}")
                callback(camera_id, None)

        threading.Thread(target=capture, daemon=True).start()


    # Table Methods
    def update_table(self, sno=None, waste_type=None, weight=None):
        def update():
            if weight is not None:
                if self.selected_row_id:
                    current_values = self.tree.item(self.selected_row_id)['values']
                    new_values = (current_values[0], current_values[1], weight)
                    self.tree.item(self.selected_row_id, values=new_values)
                    self.selected_row_id = None
            elif waste_type:
                self.selected_row_id = self.tree.insert('', 0, values=(sno, waste_type, ""))

        self.master.after(0, update)

    def save_and_clear_data(self):
        desktop_path = os.path.join(os.path.expanduser("~"), "Desktop")
        readings_folder = os.path.join(desktop_path, "AWS_MICRO", "READINGS")

        os.makedirs(readings_folder, exist_ok=True)

        filename = os.path.join(readings_folder, f"{self.credentials.get('IMG_NAME')}_{time.strftime('%Y%m%d_%H%M%S')}.json")

        current_time = time.strftime("%Y-%m-%d %H:%M:%S")
        data_to_save = {
            "RFID": self.temp_rfid_value,
            "Vehicle_No": self.temp_vehicle_no,
            "Site_Name": self.credentials.get('SITE_NAME'),
            "Timestamp": current_time,
            "Readings": []
        }

        aggregated_readings = {}
        for child in self.tree.get_children():
            values = self.tree.item(child)["values"]
            waste_type, weight = values[1], float(values[2])
            images = self.snapshot_filenames.get(waste_type, [])

            if waste_type not in aggregated_readings:
                aggregated_readings[waste_type] = {"Weight": weight, "Images": images}
            else:
                aggregated_readings[waste_type]["Weight"] += weight
                for img in images:
                    if img not in aggregated_readings[waste_type]["Images"]:
                        aggregated_readings[waste_type]["Images"].append(img)

        for waste_type, data in aggregated_readings.items():

                                     headers={'Content-Type': 'application/json'})

        except requests.RequestException as e:

            self.save_failed_data(json_data)

        self.rfid_value.set("")
        self.vehicle_no_entry.delete(0, tk.END)

    def upload_images_to_ftps(self, images, retries=3, timeout=30):
        ftp_address = self.credentials.get('FTP_ADDRESS')
        username = 'ftpuser'
        password = 'VuSvGX!htx5bgE!gNBSJ!fFE'
        ftp_directory = self.credentials.get('FTP_DIRECTORY')

        if not all([ftp_address, username, password, ftp_directory]):
            print("FTP credentials are not fully provided.")
            return

        output_folder = os.path.join(os.path.expanduser("~"), "Desktop", "AWS_MICRO", "CAMERA_OUTPUT")

        def upload_image(image):
            attempt = 0
            while attempt < retries:
                try:
                    ftps = FTP_TLS(ftp_address, timeout=timeout)
                    ftps.login(username, password)
                    ftps.prot_p()
                    ftps.cwd(ftp_directory)

                    filepath = os.path.join(output_folder, image)
                    if os.path.exists(filepath):
                        with open(filepath, 'rb') as file:
                            ftps.storbinary(f'STOR {image}', file)
                            print(f"Uploaded {image} to FTPS.")
                            ftps.quit()
                            return True
                    else:
                        print(f"File does not exist: {filepath}")
                        ftps.quit()
                        return False
                except Exception as e:
                    print(f"Failed to upload {image} on attempt {attempt + 1}/{retries}: {e}")
                    attempt += 1
                    time.sleep(5)
            return False

        batch_success = True
        with concurrent.futures.ThreadPoolExecutor(max_workers=5) as executor:
            future_to_image = {executor.submit(upload_image, image): image for image in images}
            for future in concurrent.futures.as_completed(future_to_image):
                image = future_to_image[future]
                try:
                    success = future.result()
                    if not success:
                        batch_success = False
                except Exception as e:
                    print(f"Exception occurred during upload of {image}: {e}")
                    batch_success = False

        if not batch_success:
            self.save_failed_uploads(images)

        self.master.after(0, lambda: self.update_counters_based_on_batch_success(batch_success))
