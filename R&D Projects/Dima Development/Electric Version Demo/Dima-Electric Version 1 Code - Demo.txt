/*
 * Device: PIC12F683
 * Compiler: XC8
 * Function: Vacuum Cleaner Controller (UPDATED)
 *  - Brush:
 *      * Both HANDLE1 & HANDLE2 HIGH -> Brush ON immediately
 *      * While Brush is ON: if either handle goes LOW, start 5-min OFF timer.
 *        If both handles return HIGH before 5 min, cancel timer and keep ON.
 *        If still LOW when 5 min expires, turn Brush OFF.
 *  - Vacuum:
 *      * LEVER HIGH  -> Vacuum ON immediately
 *      * LEVER LOW   -> Vacuum OFF after 10 s (cancel if lever goes HIGH again)
 *
 * Pin Map:
 *   GP5 (pin2) -> BRUSH_RELAY (to BC547 via 1k)
 *   GP0 (pin7) -> VACUUM_RELAY (to BC547 via 1k)
 *   GP1 (pin6) <- HANDLE_SW1  (10k pulldown, active HIGH)
 *   GP4 (pin3) <- HANDLE_SW2  (10k pulldown, active HIGH)
 *   GP2 (pin5) <- LEVER_SW    (10k pulldown, active HIGH)
 *   GP3/MCLR not used as I/O
 */

#include <xc.h>
#include <stdint.h>
#include <stdbool.h>

// ============================
// Configuration Bits
// ============================
#pragma config FOSC = INTRCIO, WDTE = ON, PWRTE = ON, MCLRE = ON, CP = OFF, CPD = OFF, BOREN = ON, IESO = OFF, FCMEN = OFF

// ============================
// Timing constants (ms)
// ============================
#define SYS_TICK_MS              1u
#define DEBOUNCE_MS              20u
#define VAC_OFF_DELAY_MS         10000u      // 10 seconds
#define BRUSH_OFF_DELAY_MS       300000u     // 5 minutes

// ============================
// I/O bit helpers
// ============================
#define READ_HANDLE1   ((GPIO >> 1) & 0x1)   // GP1
#define READ_HANDLE2   ((GPIO >> 4) & 0x1)   // GP4
#define READ_LEVER     ((GPIO >> 2) & 0x1)   // GP2

// ============================
// Globals updated in ISR
// ============================
volatile uint32_t g_ms = 0;

// ============================
// Debounce
// ============================
typedef struct {
    uint8_t stable;
    uint8_t last;
    uint16_t cnt;
} debounce_t;

static debounce_t db_handle1 = {0,0,0};
static debounce_t db_handle2 = {0,0,0};
static debounce_t db_lever   = {0,0,0};

static inline void debounce_update(debounce_t* db, uint8_t raw)
{
    if (raw != db->last) {
        db->last = raw;
        db->cnt  = 0;
    } else if (db->cnt < DEBOUNCE_MS) {
        db->cnt++;
        if (db->cnt >= DEBOUNCE_MS) {
            db->stable = db->last;
        }
    }
}

// ============================
// Timer1: 1 ms tick @ 8 MHz
// Fosc/4 = 2 MHz => 0.5us per tick
// Prescale 1:8 => 4us per tick; 1ms => 250 counts
// Preload = 65536 - 250 = 65286
// ============================
static void tmr1_init(void)
{
    T1CON = 0;
    T1CONbits.TMR1CS = 0;
    T1CONbits.T1CKPS = 0b11; // 1:8
    TMR1H = (uint8_t)(65286 >> 8);
    TMR1L = (uint8_t)(65286 & 0xFF);
    PIR1bits.TMR1IF = 0;
    PIE1bits.TMR1IE = 1;
    INTCONbits.PEIE = 1;
    INTCONbits.GIE  = 1;
    T1CONbits.TMR1ON = 1;
}

void __interrupt() isr(void)
{
    if (PIR1bits.TMR1IF) {
        TMR1H = (uint8_t)(65286 >> 8);
        TMR1L = (uint8_t)(65286 & 0xFF);
        PIR1bits.TMR1IF = 0;
        g_ms++;
    }
}

static inline uint32_t millis(void) { return g_ms; }

// ============================
// Clock / I/O init
// ============================
static void io_clock_init(void)
{
    OSCCONbits.IRCF = 0b111; // 8MHz
    ANSEL = 0x00;
    CMCON0 = 0x07;

    TRISIO = 0;
    TRISIObits.TRISIO5 = 0; // GP5 out (BRUSH)
    TRISIObits.TRISIO0 = 0; // GP0 out (VAC)
    TRISIObits.TRISIO1 = 1; // GP1 in  (H1)
    TRISIObits.TRISIO4 = 1; // GP4 in  (H2)
    TRISIObits.TRISIO2 = 1; // GP2 in  (LEV)

    OPTION_REGbits.nGPPU = 1; // disable weak PU (we use external pulldowns)
    WPU = 0x00;

    // Outputs default OFF (clear GP5, GP0)
    GPIO &= ~((1<<5) | (1<<0));
}

// ============================
// App state
// ============================
typedef struct {
    bool brush_on;
    bool vacuum_on;

    // Vacuum delayed OFF
    bool vac_off_pending;
    uint32_t vac_off_deadline_ms;

    // Brush delayed OFF (NEW)
    bool brush_off_pending;
    uint32_t brush_off_deadline_ms;
} app_state_t;

static app_state_t app = { false, false, false, 0, false, 0 };

static inline void set_brush(bool on)
{
    if (on) GPIO |=  (1<<5); else GPIO &= ~(1<<5); // GP5
}
static inline void set_vacuum(bool on)
{
    if (on) GPIO |=  (1<<0); else GPIO &= ~(1<<0); // GP0
}

// ============================
// Main
// ============================
void main(void)
{
    io_clock_init();
    tmr1_init();

    app.brush_on  = false;
    app.vacuum_on = false;
    app.vac_off_pending = false;
    app.brush_off_pending = false;
    set_brush(false);
    set_vacuum(false);

    for (;;) {
        // Debounce
        uint8_t raw_h1  = READ_HANDLE1 ? 1 : 0;
        uint8_t raw_h2  = READ_HANDLE2 ? 1 : 0;
        uint8_t raw_lev = READ_LEVER   ? 1 : 0;

        debounce_update(&db_handle1, raw_h1);
        debounce_update(&db_handle2, raw_h2);
        debounce_update(&db_lever,   raw_lev);

        uint8_t h1  = db_handle1.stable;
        uint8_t h2  = db_handle2.stable;
        uint8_t lev = db_lever.stable;

        uint32_t now = millis();
        bool both_high  = (h1 && h2);
        bool either_low = (!h1 || !h2);

        // ===== Brush state machine (UPDATED) =====
        // Rule:
        //   - If both handles HIGH -> Brush ON immediately; cancel pending off.
        //   - While Brush ON: if either goes LOW -> start 5-min OFF timer.
        //     If both back HIGH before deadline -> cancel timer (stay ON).
        //     If deadline reached while still LOW -> turn OFF.
        if (both_high) {
            app.brush_on = true;
            app.brush_off_pending = false;  // cancel any pending off
        } else {
            if (app.brush_on) {
                // We're running and now at least one handle is LOW -> start/continue pending off
                if (!app.brush_off_pending) {
                    app.brush_off_pending = true;
                    app.brush_off_deadline_ms = now + BRUSH_OFF_DELAY_MS;
                } else {
                    // if deadline reached and still not both high -> turn off
                    if ((int32_t)(now - app.brush_off_deadline_ms) >= 0) {
                        app.brush_on = false;
                        app.brush_off_pending = false;
                    }
                }
            } else {
                // Already OFF -> do not run the 5-min timer; wait for both_high to turn ON
                app.brush_off_pending = false;
            }
        }

        // ===== Vacuum state machine (unchanged) =====
        if (lev) {
            app.vacuum_on = true;
            app.vac_off_pending = false;
        } else {
            if (!app.vac_off_pending && app.vacuum_on) {
                app.vac_off_pending = true;
                app.vac_off_deadline_ms = now + VAC_OFF_DELAY_MS;
            }
        }
        if (app.vac_off_pending && !lev) {
            if ((int32_t)(now - app.vac_off_deadline_ms) >= 0) {
                app.vacuum_on = false;
                app.vac_off_pending = false;
            }
        }

        // Drive outputs
        set_brush(app.brush_on);
        set_vacuum(app.vacuum_on);

        // Watchdog
        CLRWDT();
    }
}
