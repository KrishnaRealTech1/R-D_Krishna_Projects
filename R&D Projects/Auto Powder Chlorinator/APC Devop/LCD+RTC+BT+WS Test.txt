/* ========================= LCD (PCF8574) ==================================== */
#define LCD_ADDR_HEX 0x27
#define LCD_ADDR     (LCD_ADDR_HEX<<1)
static uint8_t lcd_bl = 0x08;

static void lcd_i2c_byte(uint8_t v){ HAL_I2C_Master_Transmit(&hi2c1, LCD_ADDR, &v, 1, 50); }
static void lcd_write4(uint8_t nib, uint8_t rs){
    uint8_t d=((nib&0x0F)<<4)|(rs?0x01:0)|lcd_bl;
    uint8_t en=d|0x04;
    lcd_i2c_byte(en);
    lcd_i2c_byte(d);
    HAL_Delay(1);
}
static void lcd_cmd(uint8_t c){ lcd_write4(c>>4,0); lcd_write4(c&0x0F,0); }
static void lcd_data(uint8_t d){ lcd_write4(d>>4,1); lcd_write4(d&0x0F,1); }
static void lcd_init(void){
    HAL_Delay(50);
    lcd_write4(0x03,0); HAL_Delay(5);
    lcd_write4(0x03,0); HAL_Delay(5);
    lcd_write4(0x03,0); HAL_Delay(5);
    lcd_write4(0x02,0);
    lcd_cmd(0x28); lcd_cmd(0x0C); lcd_cmd(0x06);
    lcd_cmd(0x01); HAL_Delay(3);
}
static void lcd_set(uint8_t col,uint8_t row){ static const uint8_t off[]={0x00,0x40,0x14,0x54}; lcd_cmd(0x80 | (col+off[row])); }
static void lcd_clear(void){ lcd_cmd(0x01); HAL_Delay(3); }
static void lcd_print(const char* s){ while(*s) lcd_data((uint8_t)*s++); }
static void lcd_printf(const char* f, ...){
    char b[32]; va_list ap; va_start(ap,f);
    vsnprintf(b,sizeof(b),f,ap); va_end(ap);
    lcd_print(b);
}

/* ========================= UART helpers ==================================== */
static inline void bt_print(const char* s){ HAL_UART_Transmit(&huart1,(uint8_t*)s,strlen(s),100); }

/* ========================= Scale frame parser =============================== */
/* We see a lot of noise bytes (FF/FE/FD/…) then: 0x02 'K' ' ' ±NNN.NN '\r' '\n'
   We ignore everything until STX (0x02), capture printable bytes until CR.
*/
typedef enum { SF_IDLE=0, SF_IN } sstate_t;
static volatile sstate_t sst = SF_IDLE;
static volatile char sline[64];
static volatile uint8_t slen = 0;
static volatile uint8_t sready = 0;

static uint8_t rx2;

void HAL_UART_RxCpltCallback(UART_HandleTypeDef *huart)
{
    if(huart == &huart2){
        uint8_t b = rx2;

        if(sst == SF_IDLE){
            if(b == 0x02 || b == '<'){  // accept STX or '<' (some formats)
                slen = 0;
                sst  = SF_IN;
            }
        }else{ // SF_IN
            if(b == '\r' || b == '\n'){
                sline[slen] = 0;
                sready = 1;
                sst = SF_IDLE;
            }else{
                if(slen < sizeof(sline)-1){
                    if(b >= 0x20 && b <= 0x7E) sline[slen++] = (char)b; // keep printable only
                }
            }
        }

        HAL_UART_Receive_IT(&huart2, &rx2, 1);
    }
}

/* Robust number parser: looks for 'K' then a signed float after optional spaces.
   Returns true & grams_out if it parsed something sane. */
static bool parse_scale_line_to_grams(const char* line, int* grams_out)
{
    // Example payloads we've seen after STX:
    // "K -000.48"   -> -480 g  (kg)
    // "K +001.638"  -> 1638 g  (kg)
    // Also sometimes prefixes like "2>K +000.48" if you ever see '<2>'
    const char* k = strchr(line,'K');
    if(!k) return false;
    const char* p = k + 1;
    while(*p == ' ') p++;

    // signed float
    char *endp = NULL;
    float v = strtof(p, &endp);
    if(endp == p) return false; // no number

    // assume 'K' means kilograms
    float g = v * 1000.0f;
    int gi = (int)(g + (g>=0 ? 0.5f : -0.5f));
    *grams_out = gi;
    return true;
}

/* ========================= Main ============================================ */
int main(void)
{
    HAL_Init();
    SystemClock_Config();
    MX_GPIO_Init();
    MX_I2C1_Init();
    MX_USART1_UART_Init();
    MX_USART2_UART_Init();

    lcd_init();
    lcd_clear();
    lcd_set(0,0); lcd_print("Scale Parser");
    lcd_set(0,1); lcd_print("Waiting...");

    bt_print("STM32 scale parser ready\r\n");

    // Start RX on USART2
    HAL_UART_Receive_IT(&huart2, &rx2, 1);

    uint32_t t0 = HAL_GetTick();
    while(1)
    {
        // Every second print a heartbeat to BT
        if(HAL_GetTick() - t0 >= 1000){
            t0 += 1000;
            // Small heartbeat only; comment if too chatty
            bt_print("[HB]\r\n");
        }

        if(sready){
            __disable_irq();
            char buf[64];
            strncpy(buf, (const char*)sline, sizeof(buf));
            buf[sizeof(buf)-1] = 0;
            sready = 0;
            __enable_irq();

            int grams=0;
            bool ok = parse_scale_line_to_grams(buf, &grams);

            if(ok){
                char msg[64];
                snprintf(msg, sizeof(msg), "RAW:\"%s\"  G=%d\r\n", buf, grams);
                bt_print(msg);

                lcd_set(0,0); lcd_print("Weight (g)     ");
                lcd_set(0,1); lcd_printf("%d        ", grams);
            }else{
                char msg[64];
                snprintf(msg, sizeof(msg), "RAW:\"%s\" (skip)\r\n", buf);
                bt_print(msg);
            }
        }
    }
}

/* ========================= Clock / MX_* init ================================ */
void SystemClock_Config(void)
{
    RCC_OscInitTypeDef RCC_OscInitStruct={0};
    RCC_ClkInitTypeDef RCC_ClkInitStruct={0};

    __HAL_RCC_PWR_CLK_ENABLE();
    __HAL_PWR_VOLTAGESCALING_CONFIG(PWR_REGULATOR_VOLTAGE_SCALE2);

    RCC_OscInitStruct.OscillatorType=RCC_OSCILLATORTYPE_HSI;
    RCC_OscInitStruct.HSIState=RCC_HSI_ON;
    RCC_OscInitStruct.HSICalibrationValue=16;
    RCC_OscInitStruct.PLL.PLLState=RCC_PLL_ON;
    RCC_OscInitStruct.PLL.PLLSource=RCC_PLLSOURCE_HSI;
    RCC_OscInitStruct.PLL.PLLM=16;
    RCC_OscInitStruct.PLL.PLLN=336;
    RCC_OscInitStruct.PLL.PLLP=RCC_PLLP_DIV4; // 84 MHz
    RCC_OscInitStruct.PLL.PLLQ=7;
    HAL_RCC_OscConfig(&RCC_OscInitStruct);

    RCC_ClkInitStruct.ClockType=RCC_CLOCKTYPE_HCLK|RCC_CLOCKTYPE_SYSCLK|
                                RCC_CLOCKTYPE_PCLK1|RCC_CLOCKTYPE_PCLK2;
    RCC_ClkInitStruct.SYSCLKSource=RCC_SYSCLKSOURCE_PLLCLK;
    RCC_ClkInitStruct.AHBCLKDivider=RCC_SYSCLK_DIV1;
    RCC_ClkInitStruct.APB1CLKDivider=RCC_HCLK_DIV2;
    RCC_ClkInitStruct.APB2CLKDivider=RCC_HCLK_DIV1;
    HAL_RCC_ClockConfig(&RCC_ClkInitStruct, FLASH_LATENCY_2);
}

static void MX_I2C1_Init(void)
{
    hi2c1.Instance=I2C1;
    hi2c1.Init.ClockSpeed=100000;
    hi2c1.Init.DutyCycle=I2C_DUTYCYCLE_2;
    hi2c1.Init.OwnAddress1=0;
    hi2c1.Init.AddressingMode=I2C_ADDRESSINGMODE_7BIT;
    hi2c1.Init.DualAddressMode=I2C_DUALADDRESS_DISABLE;
    hi2c1.Init.OwnAddress2=0;
    hi2c1.Init.GeneralCallMode=I2C_GENERALCALL_DISABLE;
    hi2c1.Init.NoStretchMode=I2C_NOSTRETCH_DISABLE;
    HAL_I2C_Init(&hi2c1);
}

static void MX_USART1_UART_Init(void)
{
    __HAL_RCC_USART1_CLK_ENABLE();
    huart1.Instance=USART1;
    huart1.Init.BaudRate=9600;
    huart1.Init.WordLength=UART_WORDLENGTH_8B;
    huart1.Init.StopBits=UART_STOPBITS_1;
    huart1.Init.Parity=UART_PARITY_NONE;
    huart1.Init.Mode=UART_MODE_TX_RX;
    huart1.Init.HwFlowCtl=UART_HWCONTROL_NONE;
    huart1.Init.OverSampling=UART_OVERSAMPLING_16;
    HAL_UART_Init(&huart1);

    // GPIO for USART1 (PA9/PA10)
    // (GPIO clocks enabled in MX_GPIO_Init)
}

static void MX_USART2_UART_Init(void)
{
    __HAL_RCC_USART2_CLK_ENABLE();
    huart2.Instance=USART2;
    huart2.Init.BaudRate=9600;                 // scale baud
    huart2.Init.WordLength=UART_WORDLENGTH_8B;
    huart2.Init.StopBits=UART_STOPBITS_1;
    huart2.Init.Parity=UART_PARITY_NONE;
    huart2.Init.Mode=UART_MODE_TX_RX;          // RX used; TX optional
    huart2.Init.HwFlowCtl=UART_HWCONTROL_NONE;
    huart2.Init.OverSampling=UART_OVERSAMPLING_16;
    HAL_UART_Init(&huart2);

    HAL_NVIC_SetPriority(USART2_IRQn,1,0);
    HAL_NVIC_EnableIRQ(USART2_IRQn);
}

void USART2_IRQHandler(void){ HAL_UART_IRQHandler(&huart2); }

static void MX_GPIO_Init(void)
{
    __HAL_RCC_GPIOA_CLK_ENABLE();
    __HAL_RCC_GPIOB_CLK_ENABLE();

    GPIO_InitTypeDef G={0};

    // I2C1: PB8=SCL, PB9=SDA
    G.Pin=GPIO_PIN_8|GPIO_PIN_9;
    G.Mode=GPIO_MODE_AF_OD;
    G.Pull=GPIO_PULLUP;
    G.Speed=GPIO_SPEED_FREQ_VERY_HIGH;
    G.Alternate=GPIO_AF4_I2C1;
    HAL_GPIO_Init(GPIOB,&G);

    // USART1: PA9 TX, PA10 RX
    G.Pin=GPIO_PIN_9|GPIO_PIN_10;
    G.Mode=GPIO_MODE_AF_PP;
    G.Pull=GPIO_PULLUP;
    G.Speed=GPIO_SPEED_FREQ_VERY_HIGH;
    G.Alternate=GPIO_AF7_USART1;
    HAL_GPIO_Init(GPIOA,&G);

    // USART2: PA2 TX, PA3 RX
    G.Pin=GPIO_PIN_2|GPIO_PIN_3;
    G.Mode=GPIO_MODE_AF_PP;
    G.Pull=GPIO_PULLUP;
    G.Speed=GPIO_SPEED_FREQ_VERY_HIGH;
    G.Alternate=GPIO_AF7_USART2;
    HAL_GPIO_Init(GPIOA,&G);
}
