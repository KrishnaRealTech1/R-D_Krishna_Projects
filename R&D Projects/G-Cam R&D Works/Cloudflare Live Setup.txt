# For Removing Cloudeflare:

sudo bash <<'EOF'
set -euo pipefail

echo "Stopping cloudflared service if it exists..."
systemctl stop cloudflared 2>/dev/null || true
systemctl disable cloudflared 2>/dev/null || true

echo "Removing cloudflared systemd service files..."
rm -f /etc/systemd/system/cloudflared.service
rm -f /lib/systemd/system/cloudflared.service
rm -f /usr/lib/systemd/system/cloudflared.service

echo "Reloading systemd..."
systemctl daemon-reload
systemctl reset-failed cloudflared 2>/dev/null || true

echo "Removing Cloudflare Tunnel config, token, and credential files..."
rm -rf /etc/cloudflared
rm -rf /usr/local/etc/cloudflared
rm -rf /var/lib/cloudflared
rm -rf /var/log/cloudflared
rm -rf /root/.cloudflared

# Remove cloudflared config from normal users too
for home in /home/*; do
  if [ -d "$home/.cloudflared" ]; then
    echo "Removing $home/.cloudflared"
    rm -rf "$home/.cloudflared"
  fi
done

echo "Removing cloudflared package if installed through apt/dpkg..."
if dpkg -l 2>/dev/null | grep -q '^ii  cloudflared '; then
  apt-get purge -y cloudflared
  apt-get autoremove -y
else
  echo "cloudflared apt package not found."
fi

echo "Removing standalone cloudflared binaries if present..."
rm -f /usr/local/bin/cloudflared
rm -f /usr/bin/cloudflared
rm -f /bin/cloudflared

echo "Removing cloudflared user/group if present..."
if id cloudflared >/dev/null 2>&1; then
  userdel cloudflared 2>/dev/null || true
fi

if getent group cloudflared >/dev/null 2>&1; then
  groupdel cloudflared 2>/dev/null || true
fi

echo
echo "Final check:"
echo "------------"

if systemctl list-unit-files | grep -q '^cloudflared.service'; then
  echo "WARNING: cloudflared service still exists:"
  systemctl list-unit-files | grep '^cloudflared.service'
else
  echo "OK: cloudflared service removed."
fi

if command -v cloudflared >/dev/null 2>&1; then
  echo "WARNING: cloudflared binary still found at: $(command -v cloudflared)"
else
  echo "OK: cloudflared binary removed."
fi

if [ -e /etc/cloudflared ] || [ -e /root/.cloudflared ]; then
  echo "WARNING: some cloudflared config files still exist."
else
  echo "OK: cloudflared config/token files removed."
fi

echo
echo "Done. Cloudflare tunnel token, config, service, and binary have been removed where found."
EOF

#For Repair + Reinstall New Cloudflare + DNS Setup

bash <<'EOF'
set -e

echo "========================================"
echo " Cloudflare Tunnel Repair + Install"
echo "========================================"

echo
echo "[1/6] Fixing DNS..."

DEV="$(ip route show default 2>/dev/null | awk '{print $5; exit}')"

# Try to make Cloudflare/Google DNS persistent with NetworkManager
if command -v nmcli >/dev/null 2>&1 && [ -n "$DEV" ]; then
    CON="$(nmcli -g GENERAL.CONNECTION device show "$DEV" 2>/dev/null | head -n1 || true)"

    if [ -n "$CON" ] && [ "$CON" != "--" ]; then
        echo "NetworkManager connection: $CON"
        sudo nmcli connection modify "$CON" \
            ipv4.ignore-auto-dns yes \
            ipv4.dns "1.1.1.1 8.8.8.8" || true

        sudo nmcli device reapply "$DEV" 2>/dev/null || true
    fi
fi

# Try systemd-resolved
if command -v resolvectl >/dev/null 2>&1 && [ -n "$DEV" ]; then
    sudo resolvectl dns "$DEV" 1.1.1.1 8.8.8.8 2>/dev/null || true
fi

# If DNS still isn't working, temporarily repair resolv.conf
if ! getent hosts deb.debian.org >/dev/null 2>&1; then
    echo "Applying temporary /etc/resolv.conf DNS fix..."

    sudo cp -L /etc/resolv.conf \
        "/etc/resolv.conf.backup.$(date +%s)" 2>/dev/null || true

    if ! printf 'nameserver 1.1.1.1\nnameserver 8.8.8.8\n' | \
        sudo tee /etc/resolv.conf >/dev/null 2>&1; then

        sudo rm -f /etc/resolv.conf
        printf 'nameserver 1.1.1.1\nnameserver 8.8.8.8\n' | \
            sudo tee /etc/resolv.conf >/dev/null
    fi
fi

echo
echo "Testing DNS..."

if ! getent hosts pkg.cloudflare.com >/dev/null 2>&1; then
    echo
    echo "ERROR: DNS is still not working."
    echo "Check your Pi's Wi-Fi/Ethernet/router connection."
    exit 1
fi

echo "DNS OK."

echo
echo "[2/6] Installing Cloudflare repository key..."

sudo mkdir -p --mode=0755 /usr/share/keyrings

curl -fsSL https://pkg.cloudflare.com/cloudflare-main.gpg | \
    sudo tee /usr/share/keyrings/cloudflare-main.gpg >/dev/null

echo
echo "[3/6] Configuring official Cloudflare repository..."

echo 'deb [signed-by=/usr/share/keyrings/cloudflare-main.gpg] https://pkg.cloudflare.com/cloudflared any main' | \
    sudo tee /etc/apt/sources.list.d/cloudflared.list >/dev/null

echo
echo "[4/6] Updating packages..."

sudo apt-get update

echo
echo "[5/6] Repairing / reinstalling cloudflared..."

sudo apt-get install -y --reinstall cloudflared

echo
echo "Checking cloudflared..."

if ! command -v cloudflared >/dev/null 2>&1; then
    echo
    echo "ERROR: cloudflared package installed but binary still not found."
    echo
    echo "Package contents:"
    dpkg -L cloudflared
    exit 1
fi

echo
cloudflared --version

echo
echo "[6/6] Installing Cloudflare Tunnel service..."
echo
echo "Paste your NEW Cloudflare Tunnel token below."
echo "Nothing will appear while you paste/type it."
echo

IFS= read -r -s -p "NEW TUNNEL TOKEN: " TUNNEL_TOKEN </dev/tty
echo

if [ -z "$TUNNEL_TOKEN" ]; then
    echo "ERROR: No token entered."
    exit 1
fi

# Remove old service if one exists
if systemctl list-unit-files 2>/dev/null | grep -q '^cloudflared.service'; then
    echo "Removing existing cloudflared service..."
    sudo cloudflared service uninstall 2>/dev/null || true
fi

sudo cloudflared service install "$TUNNEL_TOKEN"

unset TUNNEL_TOKEN

sudo systemctl enable cloudflared >/dev/null 2>&1 || true
sudo systemctl restart cloudflared

echo
echo "========================================"
echo " SUCCESS"
echo "========================================"
echo
cloudflared --version
echo
sudo systemctl --no-pager --full status cloudflared || true

echo
echo "If you see 'active (running)' above, your tunnel is running."
EOF
