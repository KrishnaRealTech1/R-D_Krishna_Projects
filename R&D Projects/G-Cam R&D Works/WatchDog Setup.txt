#!/usr/bin/env bash

set -euo pipefail

# ============================================================
# Raspberry Pi 5 Red LED / Auto Recovery Setup Script
# ============================================================
#
# This script configures:
#   1. EEPROM power recovery settings
#   2. Raspberry Pi bootloader EEPROM update
#   3. Linux hardware watchdog
#   4. Optional GPIO heartbeat service on GPIO26
#   5. Basic power diagnostics
#
# Target:
#   Raspberry Pi 5
#   Raspberry Pi OS Lite 64-bit
#
# Important:
#   This script cannot fix a bad power supply.
#   For Pi 5, use a stable 5V 5A / official 27W USB-C power supply.
#
# GPIO used for external heartbeat:
#   BCM GPIO26 = physical pin 37
#
# ============================================================

SCRIPT_NAME="pi5-red-led-recovery-setup"
BOOT_CONFIG="/boot/firmware/config.txt"
SYSTEMD_CONFIG="/etc/systemd/system.conf"
EEPROM_BACKUP_DIR="/root/pi5-eeprom-backups"
TMP_EEPROM_CONFIG="/tmp/pi5-bootloader-config.txt"
HEARTBEAT_SCRIPT="/usr/local/bin/pi-heartbeat.py"
HEARTBEAT_SERVICE="/etc/systemd/system/pi-heartbeat.service"
HEARTBEAT_GPIO="26"

log() {
    echo
    echo "============================================================"
    echo "$1"
    echo "============================================================"
}

info() {
    echo "[INFO] $1"
}

warn() {
    echo "[WARN] $1"
}

fail() {
    echo "[ERROR] $1"
    exit 1
}

require_root() {
    if [[ "${EUID}" -ne 0 ]]; then
        fail "Please run this script with sudo: sudo bash ${SCRIPT_NAME}.sh"
    fi
}

require_command() {
    local cmd="$1"

    if ! command -v "${cmd}" >/dev/null 2>&1; then
        fail "Required command not found: ${cmd}"
    fi
}

detect_pi_model() {
    local model_file="/proc/device-tree/model"

    if [[ -f "${model_file}" ]]; then
        local model
        model="$(tr -d '\0' < "${model_file}")"
        info "Detected board: ${model}"

        if [[ "${model}" != *"Raspberry Pi 5"* ]]; then
            warn "This does not look like a Raspberry Pi 5."
            warn "Continuing anyway, but EEPROM options may not apply."
        fi
    else
        warn "Could not detect Raspberry Pi model."
    fi
}

backup_file() {
    local file="$1"

    if [[ -f "${file}" ]]; then
        cp "${file}" "${file}.backup.$(date +%Y%m%d-%H%M%S)"
        info "Backup created: ${file}.backup.*"
    fi
}

set_or_add_config_line() {
    local file="$1"
    local key="$2"
    local value="$3"
    local line="${key}=${value}"

    touch "${file}"

    if grep -qE "^[#[:space:]]*${key}=" "${file}"; then
        sed -i "s|^[#[:space:]]*${key}=.*|${line}|g" "${file}"
    else
        echo "${line}" >> "${file}"
    fi
}

update_packages() {
    log "Updating Raspberry Pi OS packages"

    apt update
    apt full-upgrade -y
}

install_required_packages() {
    log "Installing required packages"

    apt install -y python3 python3-gpiozero
}

update_eeprom_package() {
    log "Checking Raspberry Pi EEPROM tools"

    require_command rpi-eeprom-update
    require_command rpi-eeprom-config
    require_command vcgencmd

    info "Current EEPROM status:"
    rpi-eeprom-update || true

    info "Scheduling EEPROM firmware update if available..."
    rpi-eeprom-update -a || true
}

configure_eeprom_power_recovery() {
    log "Configuring EEPROM power recovery settings"

    mkdir -p "${EEPROM_BACKUP_DIR}"

    local backup_file_path="${EEPROM_BACKUP_DIR}/bootloader-config.$(date +%Y%m%d-%H%M%S).txt"

    info "Backing up current EEPROM config to: ${backup_file_path}"
    rpi-eeprom-config > "${backup_file_path}"

    cp "${backup_file_path}" "${TMP_EEPROM_CONFIG}"

    info "Setting WAIT_FOR_POWER_BUTTON=0"
    set_or_add_config_line "${TMP_EEPROM_CONFIG}" "WAIT_FOR_POWER_BUTTON" "0"

    info "Setting POWER_OFF_ON_HALT=0"
    set_or_add_config_line "${TMP_EEPROM_CONFIG}" "POWER_OFF_ON_HALT" "0"

    info "New EEPROM config to apply:"
    echo "------------------------------------------------------------"
    cat "${TMP_EEPROM_CONFIG}"
    echo "------------------------------------------------------------"

    info "Applying EEPROM config. This schedules the EEPROM update for next reboot."
    rpi-eeprom-config --apply "${TMP_EEPROM_CONFIG}"
}

configure_kernel_watchdog() {
    log "Configuring Raspberry Pi kernel watchdog"

    if [[ ! -f "${BOOT_CONFIG}" ]]; then
        fail "Boot config not found: ${BOOT_CONFIG}"
    fi

    backup_file "${BOOT_CONFIG}"

    info "Setting kernel_watchdog_timeout=15 in ${BOOT_CONFIG}"
    set_or_add_config_line "${BOOT_CONFIG}" "kernel_watchdog_timeout" "15"
}

configure_systemd_watchdog() {
    log "Configuring systemd watchdog"

    if [[ ! -f "${SYSTEMD_CONFIG}" ]]; then
        fail "systemd config not found: ${SYSTEMD_CONFIG}"
    fi

    backup_file "${SYSTEMD_CONFIG}"

    info "Setting RuntimeWatchdogSec=10"
    set_or_add_config_line "${SYSTEMD_CONFIG}" "RuntimeWatchdogSec" "10"

    info "Setting RebootWatchdogSec=10min"
    set_or_add_config_line "${SYSTEMD_CONFIG}" "RebootWatchdogSec" "10min"

    systemctl daemon-reexec
}

create_heartbeat_service() {
    log "Creating GPIO heartbeat service on GPIO${HEARTBEAT_GPIO}"

    cat > "${HEARTBEAT_SCRIPT}" <<PYTHON_EOF
#!/usr/bin/env python3

import signal
import sys
import time
from gpiozero import LED

HEARTBEAT_GPIO = ${HEARTBEAT_GPIO}
ON_TIME_SECONDS = 0.5
OFF_TIME_SECONDS = 0.5

running = True


def handle_exit(signum, frame):
    global running
    running = False


def main():
    signal.signal(signal.SIGTERM, handle_exit)
    signal.signal(signal.SIGINT, handle_exit)

    heartbeat = LED(HEARTBEAT_GPIO)

    try:
        while running:
            heartbeat.on()
            time.sleep(ON_TIME_SECONDS)
            heartbeat.off()
            time.sleep(OFF_TIME_SECONDS)
    finally:
        heartbeat.off()
        heartbeat.close()


if __name__ == "__main__":
    try:
        main()
    except Exception as exc:
        print(f"Heartbeat service failed: {exc}", file=sys.stderr)
        sys.exit(1)
PYTHON_EOF

    chmod +x "${HEARTBEAT_SCRIPT}"

    cat > "${HEARTBEAT_SERVICE}" <<SERVICE_EOF
[Unit]
Description=Raspberry Pi GPIO${HEARTBEAT_GPIO} heartbeat for external watchdog
After=multi-user.target

[Service]
Type=simple
ExecStart=/usr/bin/python3 ${HEARTBEAT_SCRIPT}
Restart=always
RestartSec=2

[Install]
WantedBy=multi-user.target
SERVICE_EOF

    systemctl daemon-reload
    systemctl enable pi-heartbeat.service
    systemctl restart pi-heartbeat.service

    info "Heartbeat service enabled and started on GPIO${HEARTBEAT_GPIO}."
}

print_diagnostics() {
    log "Diagnostics"

    info "EEPROM update status:"
    rpi-eeprom-update || true

    echo
    info "Current EEPROM config power-related lines:"
    rpi-eeprom-config | grep -E "WAIT_FOR_POWER_BUTTON|POWER_OFF_ON_HALT|WAKE_ON_GPIO" || true

    echo
    info "Watchdog devices:"
    if command -v wdctl >/dev/null 2>&1; then
        wdctl || true
    else
        warn "wdctl command not found."
    fi

    echo
    info "Systemd watchdog settings:"
    systemctl show | grep -i watchdog || true

    echo
    info "Power throttling status:"
    vcgencmd get_throttled || true

    echo
    info "Pi 5 PMIC EXT5V voltage:"
    vcgencmd pmic_read_adc EXT5V_V || true

    echo
    info "Heartbeat service status:"
    systemctl --no-pager --full status pi-heartbeat.service || true
}

main() {
    require_root

    log "Starting Raspberry Pi 5 red LED recovery setup"

    detect_pi_model
    update_packages
    install_required_packages
    update_eeprom_package
    configure_eeprom_power_recovery
    configure_kernel_watchdog
    configure_systemd_watchdog
    create_heartbeat_service
    print_diagnostics

    log "Setup complete"

    echo "IMPORTANT:"
    echo "1. Reboot is required now."
    echo "2. EEPROM changes are applied during reboot."
    echo "3. After reboot, test by removing and reconnecting power."
    echo "4. If Pi still stays red, the issue is probably power supply or hardware-level standby."
    echo
    echo "Run this now:"
    echo
    echo "    sudo reboot"
    echo
}

main "$@"

chmod +x pi5-red-led-recovery-setup.sh
sudo ./pi5-red-led-recovery-setup.sh
sudo reboot
