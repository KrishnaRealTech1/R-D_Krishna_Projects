@'
from PIL import Image, ImageOps
from pathlib import Path

SOURCE = Path("MaterialAI_Logo.png")
OUTPUT = Path("MaterialAI.ico")

sizes = [(256, 256), (128, 128), (64, 64), (48, 48), (32, 32), (16, 16)]

img = Image.open(SOURCE).convert("RGBA")

# Put the logo into a clean square transparent canvas.
# This avoids stretched or cropped EXE icons.
canvas_size = 1024
canvas = Image.new("RGBA", (canvas_size, canvas_size), (0, 0, 0, 0))

img.thumbnail((canvas_size, canvas_size), Image.Resampling.LANCZOS)

x = (canvas_size - img.width) // 2
y = (canvas_size - img.height) // 2
canvas.paste(img, (x, y), img)

canvas.save(OUTPUT, format="ICO", sizes=sizes)

print(f"Created {OUTPUT.resolve()}")
print("ICO sizes:", sizes)
