def now_string() -> str:
    return datetime.now().strftime("%Y-%m-%d_%H-%M-%S")

def display_time() -> str:
    return datetime.now().strftime("%Y-%m-%d %H:%M:%S")

def folder_name_for_material(name: str) -> str:
    if name in DEFAULT_FOLDER_NAMES:
        return DEFAULT_FOLDER_NAMES[name]
    cleaned = re.sub(r"[^A-Za-z0-9._ -]+", "", name.strip())
    cleaned = re.sub(r"\s+", "_", cleaned).strip("._- ")
    return cleaned or "Material"

def safe_filename(text: str) -> str:
    cleaned = re.sub(r"[^A-Za-z0-9._-]+", "_", text.strip()).strip("._-")
    return cleaned or "material"

def normalize_material_name(name: str) -> str:
    cleaned = name.strip().lower().replace("_", " ").replace("-", " ")
    mapping = {
        "raw material": "Raw Material",
        "6mm jalli": "6mm Jalli",
        "12mm jalli": "12mm Jalli",
        "20mm jalli": "20mm Jalli",
        "35mm jalli": "35mm Jalli",
        "m sand": "M-Sand",
        "msand": "M-Sand",
        "p sand": "P-Sand",
        "psand": "P-Sand",
        "wet mix": "Wet Mix",
        "empty": "Empty",
        "other": "Other",
    }
    return mapping.get(cleaned, name.strip() or "Other")

def read_labels() -> List[str]:
    if not LABELS_PATH.exists():
        # Write defaults on first run
        LABELS_PATH.parent.mkdir(parents=True, exist_ok=True)
        LABELS_PATH.write_text("\n".join(DEFAULT_MATERIALS), encoding="utf-8")
        return DEFAULT_MATERIALS[:]
    labels = [
        line.strip()
        for line in LABELS_PATH.read_text(encoding="utf-8").splitlines()
        if line.strip()
    ]
    return labels or DEFAULT_MATERIALS[:]

def append_label(name: str) -> None:
    labels = read_labels()
    if name in labels:
        return
    with LABELS_PATH.open("a", encoding="utf-8") as f:
        if LABELS_PATH.stat().st_size > 0:
            f.write("\n")
        f.write(name)

def resize_keep_aspect(image: np.ndarray, width: int) -> np.ndarray:
    if image is None:
        return image
    h, w = image.shape[:2]
    if w <= width:
        return image
    return cv2.resize(image, (width, int(h * width / w)), interpolation=cv2.INTER_AREA)

def encode_jpeg(image: np.ndarray, quality: int = 85) -> bytes:
    ok, buf = cv2.imencode(".jpg", image, [int(cv2.IMWRITE_JPEG_QUALITY), int(quality)])
    return buf.tobytes() if ok else b""

def save_image(image: np.ndarray, directory: Path, prefix: str) -> Path:
    directory.mkdir(parents=True, exist_ok=True)
    path = directory / f"{safe_filename(prefix)}_{now_string()}.jpg"
    if not cv2.imwrite(str(path), image, [int(cv2.IMWRITE_JPEG_QUALITY), IMAGE_JPEG_QUALITY]):
        raise RuntimeError(f"Failed to write image: {path}")
    return path

def write_text_file(directory: Path, prefix: str, content: str) -> Path:
    directory.mkdir(parents=True, exist_ok=True)
    path = directory / f"{safe_filename(prefix)}_{now_string()}.txt"
    path.write_text(content, encoding="utf-8")
    return path

def placeholder_frame(message: str) -> np.ndarray:

@dataclass
class Material:
    name:   str
    slug:   str
    folder: Path

class MaterialRegistry:
    def __init__(self) -> None:
        self.lock = threading.Lock()
        self.materials: List[Material] = []
        self.reload()

    def reload(self) -> None:
        materials: List[Material] = []
        seen_names = set()
        for label in read_labels():
            name = normalize_material_name(label)
            if name in seen_names:
                continue
            folder = REFERENCE_DIR / folder_name_for_material(name)
            folder.mkdir(parents=True, exist_ok=True)
            materials.append(Material(name=name, slug=folder.name, folder=folder))
            seen_names.add(name)
        for folder in sorted(p for p in REFERENCE_DIR.iterdir() if p.is_dir()):
            if folder.name not in {m.slug for m in materials}:
                materials.append(Material(
                    name=folder.name.replace("_", " "),
                    slug=folder.name,
                    folder=folder,
                ))
        with self.lock:
            self.materials = materials

    def list(self) -> List[Material]:
        with self.lock:
            return list(self.materials)

    def get_by_slug(self, slug: str) -> Optional[Material]:
        with self.lock:
            for m in self.materials:
                if m.slug == slug:
                    return m
        return None

    def add(self, name: str) -> Material:
        name = re.sub(r"\s+", " ", name.strip())
        if not name:
            raise ValueError("Material name is required")
        if len(name) > 60:
            raise ValueError("Material name is too long (max 60 chars)")
        if not re.match(r"^[A-Za-z0-9][A-Za-z0-9 ._/-]*$", name):
            raise ValueError("Use letters, numbers, spaces, dot, slash, underscore or hyphen only")
        folder = REFERENCE_DIR / folder_name_for_material(name)
        folder.mkdir(parents=True, exist_ok=True)
        append_label(name)
        self.reload()
        material = self.get_by_slug(folder.name)
        if material is None:
            raise RuntimeError("Material created but not found after reload")
        return material

registry = MaterialRegistry()

# ════════════════════════════════════════════════════════════
#   IMAGE SIMILARITY (Reference matching)
# ════════════════════════════════════════════════════════════

def color_hist_similarity(img1: np.ndarray, img2: np.ndarray) -> float:
    try:
        img1 = resize_keep_aspect(img1, IMAGE_WIDTH_FOR_PROCESSING)
        img2 = resize_keep_aspect(img2, IMAGE_WIDTH_FOR_PROCESSING)
        hsv1 = cv2.cvtColor(img1, cv2.COLOR_BGR2HSV)
        hsv2 = cv2.cvtColor(img2, cv2.COLOR_BGR2HSV)
        h1 = cv2.calcHist([hsv1], [0,1,2], None, [24,24,24], [0,180,0,256,0,256])
        h2 = cv2.calcHist([hsv2], [0,1,2], None, [24,24,24], [0,180,0,256,0,256])
        cv2.normalize(h1, h1)
        cv2.normalize(h2, h2)
        return max(0.0, min(1.0, float(cv2.compareHist(h1, h2, cv2.HISTCMP_CORREL))))
    except Exception:
        return 0.0

def texture_similarity(img1: np.ndarray, img2: np.ndarray) -> float:
    try:
        g1 = cv2.resize(cv2.cvtColor(resize_keep_aspect(img1, IMAGE_WIDTH_FOR_PROCESSING),
                                      cv2.COLOR_BGR2GRAY), (320,240), interpolation=cv2.INTER_AREA)
        g2 = cv2.resize(cv2.cvtColor(resize_keep_aspect(img2, IMAGE_WIDTH_FOR_PROCESSING),
                                      cv2.COLOR_BGR2GRAY), (320,240), interpolation=cv2.INTER_AREA)
        lap1 = cv2.Laplacian(g1, cv2.CV_32F)
        lap2 = cv2.Laplacian(g2, cv2.CV_32F)
        h1 = cv2.calcHist([np.uint8(np.absolute(lap1))], [0], None, [64], [0,256])
        h2 = cv2.calcHist([np.uint8(np.absolute(lap2))], [0], None, [64], [0,256])
        cv2.normalize(h1, h1)
        cv2.normalize(h2, h2)
        return max(0.0, min(1.0, float(cv2.compareHist(h1, h2, cv2.HISTCMP_CORREL))))
    except Exception:
        return 0.0

def orb_similarity(img1: np.ndarray, img2: np.ndarray) -> float:
    try:
        g1 = cv2.cvtColor(resize_keep_aspect(img1, IMAGE_WIDTH_FOR_PROCESSING), cv2.COLOR_BGR2GRAY)
        g2 = cv2.cvtColor(resize_keep_aspect(img2, IMAGE_WIDTH_FOR_PROCESSING), cv2.COLOR_BGR2GRAY)
        orb = cv2.ORB_create(nfeatures=400)
        kp1, des1 = orb.detectAndCompute(g1, None)
        kp2, des2 = orb.detectAndCompute(g2, None)
        if des1 is None or des2 is None or len(kp1) < 8 or len(kp2) < 8:
            return 0.0
        matches = cv2.BFMatcher(cv2.NORM_HAMMING, crossCheck=True).match(des1, des2)
        good = [m for m in matches if m.distance < 62]
        return max(0.0, min(1.0, len(good) / float(max(1, min(len(kp1), len(kp2))))))
    except Exception:
        return 0.0

def combined_reference_score(captured: np.ndarray, reference: np.ndarray) -> float:
    c = color_hist_similarity(captured, reference)
    t = texture_similarity(captured, reference)
    o = orb_similarity(captured, reference)
    return max(0.0, min(1.0, 0.50*c + 0.30*t + 0.20*o))

#   REFERENCE IMAGE STORE
# ════════════════════════════════════════════════════════════

class ReferenceStore:
    def __init__(self) -> None:
        self.lock = threading.Lock()
        self.references: Dict[str, List[Tuple[Path, np.ndarray]]] = {}
        self.reload()

    def reload(self) -> None:
        loaded: Dict[str, List[Tuple[Path, np.ndarray]]] = {}
        for material in registry.list():
            images = []
            if material.folder.exists():
                for path in sorted(material.folder.iterdir(), key=lambda p: p.stat().st_mtime):
                    if path.suffix.lower() in IMAGE_EXTENSIONS:
                        img = cv2.imread(str(path))
                        if img is not None:
                            images.append((path, img))
            loaded[material.name] = images
        with self.lock:
            self.references = loaded
        logger.info("Loaded %d reference images", sum(len(v) for v in loaded.values()))

    def best_match(self, image: np.ndarray) -> Tuple[str, float]:
        best_label, best_score = "Other", 0.0
        with self.lock:
            items = [(mat, refs[:]) for mat, refs in self.references.items()]
        for material, refs in items:
            for _, ref_img in refs:
                score = combined_reference_score(image, ref_img)
                if score > best_score:
                    best_score = score
                    best_label = material
        return best_label, best_score

    def metadata(self) -> List[Dict]:
        result = []
        for material in registry.list():
            paths = sorted(
                [p for p in material.folder.iterdir()
                 if p.is_file() and p.suffix.lower() in IMAGE_EXTENSIONS]
                if material.folder.exists() else [],
                key=lambda p: p.stat().st_mtime,
                reverse=True,
            )
            latest = paths[0] if paths else None
            result.append({
                "name":         material.name,
                "slug":         material.slug,
                "count":        len(paths),
                "latest_name":  latest.name if latest else None,
                "latest_mtime": int(latest.stat().st_mtime) if latest else 0,
                "latest_url":   f"/reference/{material.slug}/{latest.name}" if latest else None,
            })
        return result

reference_store = ReferenceStore()

class CameraStream:
    def __init__(self, source) -> None:
        self.source = source
        self.lock = threading.Lock()
        self.frame: Optional[np.ndarray] = None
        self.running = False

    def start(self) -> None:
        if self.running:
            return
        self.running = True
        threading.Thread(target=self._reader, daemon=True).start()

    def _reader(self) -> None:
        while self.running:
            cap = None
            try:
                logger.info("Opening camera source: %s", self.source)
                cap = cv2.VideoCapture(self.source)
                if not cap.isOpened():
                    state_store.update(camera_connected=False, last_error="Camera open failed")
                    time.sleep(RTSP_RECONNECT_SECONDS)
                    continue
                state_store.update(camera_connected=True, last_error=None)
                while self.running:
                    ok, frame = cap.read()
                    if not ok or frame is None:
                        state_store.update(camera_connected=False, last_error="Camera read failed")
                        break
                    with self.lock:
                        self.frame = frame.copy()
                    state_store.update(camera_connected=True, last_frame_at=display_time())
                    time.sleep(0.01)
            except Exception as e:
                logger.error("Camera error: %s", e)
                state_store.update(camera_connected=False, last_error=str(e))
            finally:
                if cap is not None:
                    cap.release()
            time.sleep(RTSP_RECONNECT_SECONDS)

    def latest(self) -> Optional[np.ndarray]:
        with self.lock:
            return None if self.frame is None else self.frame.copy()

    def capture(self) -> Optional[np.ndarray]:
        frame = self.latest()
        if frame is not None:
            return frame
        # Direct single capture attempt
        cap = cv2.VideoCapture(self.source)
        if not cap.isOpened():
            return None
        try:
            captured = None
            for _ in range(20):
                ok, temp = cap.read()
                if ok and temp is not None:
                    captured = temp
                time.sleep(0.04)
            return captured
        finally:
            cap.release()

class TFLiteMaterialClassifier:
    def __init__(self) -> None:
        self.available = False
        self.interpreter = None
        self.input_details = None
        self.output_details = None
        self.labels = read_labels()

        if Interpreter is None:
            logger.warning("TFLite runtime not available — AI model disabled")
            return
        if not MODEL_PATH.exists():
            logger.warning("TFLite model not found at %s — AI model disabled", MODEL_PATH)
            return
        try:
            self.interpreter = Interpreter(model_path=str(MODEL_PATH))
            self.interpreter.allocate_tensors()
            self.input_details  = self.interpreter.get_input_details()
            self.output_details = self.interpreter.get_output_details()
            self.available = True
            logger.info("TFLite model loaded OK")
        except Exception as e:
            logger.error("TFLite load failed: %s", e)

    def predict(self, image_bgr: np.ndarray) -> Tuple[str, float]:
        if not self.available:
            return "Other", 0.0
        try:
            inp = self.input_details[0]
            h, w = int(inp["shape"][1]), int(inp["shape"][2])
            rgb     = cv2.cvtColor(image_bgr, cv2.COLOR_BGR2RGB)
            resized = cv2.resize(rgb, (w, h), interpolation=cv2.INTER_AREA)
            if inp["dtype"] == np.float32:
                data = resized.astype(np.float32) / 255.0
            else:
                data = resized.astype(inp["dtype"])
            data = np.expand_dims(data, axis=0)
            self.interpreter.set_tensor(inp["index"], data)
            self.interpreter.invoke()
            out_detail = self.output_details[0]
            output = self.interpreter.get_tensor(out_detail["index"])[0]
            if output.dtype != np.float32:
                scale, zero_point = out_detail.get("quantization", (0.0, 0))
                output = (scale * (output.astype(np.float32) - zero_point)) if scale else output.astype(np.float32)
            if np.max(output) > 1.0 or np.min(output) < 0.0:
                exp = np.exp(output - np.max(output))
                probs = exp / np.sum(exp)
            else:
                probs = output
            idx = int(np.argmax(probs))
            label = self.labels[idx] if idx < len(self.labels) else "Other"
            return normalize_material_name(label), float(probs[idx])
        except Exception as e:
            logger.error("AI predict failed: %s", e)
            return "Other", 0.0

def _unique(values: List[str]) -> List[str]:
    seen, out = set(), []
    for v in values:
        v = v.strip().rstrip("/")
        if v and v not in seen:
            seen.add(v)
            out.append(v)
    return out

def _sftp_candidates(remote_dir: str) -> List[str]:
    raw = remote_dir.strip().rstrip("/") or "/"
    candidates = [raw]
    for prefix in [f"/home/{SFTP_USERNAME}/", "/home/ftpuser/"]:
        if raw.startswith(prefix):
            rel = raw[len(prefix):].strip("/")
            if rel:
                candidates += [rel, "/" + rel]
    stripped = raw.strip("/")
    if stripped:
        candidates.append(stripped)
    return _unique(candidates)

def _sftp_mkdir_p(sftp: paramiko.SFTPClient, remote_dir: str) -> None:
    remote_dir = remote_dir.rstrip("/")
    if not remote_dir or remote_dir == ".":
        return
    if remote_dir.startswith("/"):
        current = "/"
        parts = [p for p in remote_dir.split("/") if p]
    else:
        current = "."
        parts = [p for p in remote_dir.split("/") if p]
    for part in parts:
        current = f"/{part}" if current == "/" else (part if current == "." else f"{current}/{part}")
        try:
            sftp.stat(current)
        except IOError:
            sftp.mkdir(current)

def upload_to_sftp(path: Path) -> Tuple[bool, str, Optional[str]]:
    if not SFTP_HOST or not SFTP_USERNAME:
        msg = "SFTP not configured"
        state_store.update(last_sftp_ok=False, last_sftp_status=msg, last_error=msg)
        return False, msg, None

    client = sftp = None
    errors = []
    try:
        state_store.update(last_sftp_status="Connecting", last_error=None)
        client = paramiko.SSHClient()
        client.set_missing_host_key_policy(paramiko.AutoAddPolicy())
        client.connect(
            hostname=SFTP_HOST, port=SFTP_PORT,
            username=SFTP_USERNAME, password=SFTP_PASSWORD,
            timeout=30, banner_timeout=30, auth_timeout=30,
            look_for_keys=False, allow_agent=False,
        )
        sftp = client.open_sftp()

        for remote_dir in _sftp_candidates(SFTP_REMOTE_DIR):
            try:
                state_store.update(last_sftp_status=f"Trying {remote_dir}")
                _sftp_mkdir_p(sftp, remote_dir)
                remote_path = remote_dir.rstrip("/") + "/" + path.name
                tmp_path    = remote_path + ".uploading"
                sftp.put(str(path), tmp_path)
                sftp.rename(tmp_path, remote_path)
                sftp.stat(remote_path)
                state_store.update(last_sftp_ok=True, last_sftp_status="Uploaded",
                                   last_sftp_remote_path=remote_path, last_error=None)
                logger.info("SFTP uploaded: %s -> %s", path, remote_path)
                return True, "Uploaded", remote_path
            except Exception as e:
                errors.append(f"{remote_dir}: {e}")
                logger.warning("SFTP candidate failed: %s: %s", remote_dir, e)

        msg = "SFTP upload failed: " + " | ".join(errors)
        logger.error(msg)
        state_store.update(last_sftp_ok=False, last_sftp_status="Failed", last_error=msg)
        try:
            FAILED_DIR.mkdir(parents=True, exist_ok=True)
            dest = FAILED_DIR / path.name
            if path.exists() and dest != path:
                dest.write_bytes(path.read_bytes())
        except Exception:
            pass
        return False, msg, None

    except Exception as e:
        msg = f"SFTP connection failed: {e}"
        logger.error(msg)
        state_store.update(last_sftp_ok=False, last_sftp_status="Connection Failed", last_error=msg)
        return False, msg, None
    finally:
        try:
            if sftp:   sftp.close()
            if client: client.close()
        except Exception:
            pass

#   UART / SERIAL OUTPUT

def send_uart(material: str, ai_conf: float, ref_score: float) -> bool:
    if not UART_ENABLED:
        state_store.update(last_uart_ok=False, last_uart_status="Disabled")
        return False
    if not SERIAL_AVAILABLE:
        state_store.update(last_uart_ok=False, last_uart_status="pyserial missing")
        return False
    msg = f"MATERIAL:{material};AI:{ai_conf:.3f};REF:{ref_score:.3f}\n"
    try:
        with serial.Serial(UART_PORT, UART_BAUDRATE, timeout=2) as port:
            port.write(msg.encode("utf-8"))
            port.flush()
        logger.info("UART sent: %s", msg.strip())
        state_store.update(last_uart_ok=True, last_uart_status="Sent")
        return True
    except Exception as e:
        err = f"UART failed: {e}"
        logger.error(err)
        state_store.update(last_uart_ok=False, last_uart_status="Failed", last_error=err)
        return False

# ════════════════════════════════════════════════════════════
#   CLASSIFICATION DECISION
# ════════════════════════════════════════════════════════════

def decide_material(ai_label: str, ai_conf: float,
                    ref_label: str, ref_score: float) -> Tuple[str, str]:
    if ai_conf >= AI_CONFIDENCE_THRESHOLD and ai_label != "Other":
        return ai_label, "AI Model"
    if ref_score >= REFERENCE_CONFIDENCE_THRESHOLD:
        return ref_label, "Reference Image"
    if ai_conf > 0:
        return ai_label, "Low Confidence AI"
    return "Other", "Fallback"

def process_captured_frame(frame: np.ndarray, source: str = "manual") -> Dict:
    state_store.update(sensor_status=f"Processing ({source})", last_error=None)

    ai_label, ai_conf    = classifier.predict(frame)
    ref_label, ref_score = reference_store.best_match(frame)
    material, decision   = decide_material(ai_label, ai_conf, ref_label, ref_score)

    saved_path = save_image(frame, CAPTURE_DIR, material)

    sftp_ok, sftp_status, remote_path = upload_to_sftp(saved_path)
    uart_ok = send_uart(material, ai_conf, ref_score)

    if sftp_ok and DELETE_IMAGE_AFTER_SUCCESS:
        try:
            saved_path.unlink(missing_ok=True)
        except Exception:
            pass

    state_store.update(
        sensor_status       = "Ready — waiting for trigger",
        last_material       = material,
        decision_source     = decision,
        ai_label            = ai_label,
        ai_confidence       = ai_conf,
        reference_label     = ref_label,
        reference_score     = ref_score,
        last_capture_url    = f"/captured/{saved_path.name}",
        last_sftp_ok        = sftp_ok,
        last_sftp_status    = "Uploaded" if sftp_ok else "Failed",
        last_sftp_remote_path = remote_path,
        last_uart_ok        = uart_ok,
        last_uart_status    = "Sent" if uart_ok else "Failed",
        last_error          = None if sftp_ok else sftp_status,
    )

    logger.info("Detected: %s  src=%s  ai=%s(%.3f)  ref=%s(%.3f)  sftp=%s",
                material, decision, ai_label, ai_conf, ref_label, ref_score, sftp_ok)

    return {
        "material": material, "decision_source": decision,
        "ai_label": ai_label, "ai_confidence": ai_conf,
        "reference_label": ref_label, "reference_score": ref_score,
        "image": saved_path.name,
        "sftp_ok": sftp_ok, "sftp_status": sftp_status, "remote_path": remote_path,
        "uart_ok": uart_ok,
    }

#   TRIGGER ENGINE  (replaces Pi GPIO sensor)

_trigger_lock    = threading.Lock()
_last_trigger_at = 0.0

def fire_trigger(source: str = "manual") -> Optional[Dict]:
    global _last_trigger_at
    now = time.time()
    with _trigger_lock:
        if now - _last_trigger_at < DEBOUNCE_SECONDS:
            return None
        _last_trigger_at = now

    logger.info("Trigger fired from: %s — waiting %.1fs", source, CAPTURE_DELAY_SECONDS)
    state_store.update(sensor_status=f"Triggered ({source}) — waiting {CAPTURE_DELAY_SECONDS:g}s")
    time.sleep(CAPTURE_DELAY_SECONDS)

    frame = camera_stream.capture()
    if frame is None:
        state_store.update(sensor_status="Capture failed", last_error="No frame captured")
        logger.error("No frame captured after trigger")
        return None

    try:
        return process_captured_frame(frame, source=source)
    except Exception as e:
        logger.error("Processing failed: %s", e)
        logger.error(traceback.format_exc())
        state_store.update(sensor_status="Process error", last_error=str(e))
        return None

def _keyboard_trigger_thread() -> None:
    if not KEYBOARD_AVAILABLE or not TRIGGER_HOTKEY:
        return
    logger.info("Keyboard trigger active — press  '%s'  to capture", TRIGGER_HOTKEY)
    state_store.update(sensor_status=f"Ready — press [{TRIGGER_HOTKEY.upper()}] or use dashboard button")
    try:
        _kb.add_hotkey(TRIGGER_HOTKEY, lambda: threading.Thread(
            target=fire_trigger, args=("keyboard",), daemon=True).start())
        _kb.wait()
    except Exception as e:
        logger.error("Keyboard listener error: %s", e)

@app.route("/api/material/<slug>/upload", methods=["POST"])
def api_upload_reference(slug: str):
    material = registry.get_by_slug(slug)
    if material is None:
        return jsonify(ok=False, error="Material not found"), 404
    uploaded = request.files.get("image")
    if not uploaded or not uploaded.filename:
        return jsonify(ok=False, error="Image file required"), 400
    if Path(uploaded.filename).suffix.lower() not in IMAGE_EXTENSIONS:
        return jsonify(ok=False, error="Only JPG, PNG, BMP allowed"), 400
    data = np.frombuffer(uploaded.read(), np.uint8)
    image = cv2.imdecode(data, cv2.IMREAD_COLOR)
    if image is None:
        return jsonify(ok=False, error="Could not decode image"), 400
    try:
        path = save_image(image, material.folder, f"reference_{material.slug}")
        reference_store.reload()
        return jsonify(ok=True, file=path.name)
    except Exception as e:
        return jsonify(ok=False, error=str(e)), 500
